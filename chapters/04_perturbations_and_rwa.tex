\chapter{Driven systems: perturbation theory and the rotating-wave approximation}
\label{ch:perturbations}


The Schr\"odinger equation can rarely be solved exactly when the
Hamiltonian is time-dependent. Two systematic approximations carry us
through almost every situation we will meet later. The first is
\emph{time-dependent perturbation theory}, in which a small coupling
$\op{V}(t)$ added to a known $\op{H}_0$ generates transitions whose
amplitudes are computed order by order; specialising to a continuum
of final states gives Fermi's golden rule, the standard expression
for transition rates. The second is the \emph{rotating-wave
approximation} (RWA), in which fast counter-rotating terms in the
interaction-picture Hamiltonian are dropped because they average to
zero on the timescales of interest. Both approximations are
ingredients of the Jaynes--Cummings model
(\cref{ch:jaynes-cummings}), of dispersive readout
(\cref{sec:dispersive}), and of every drive-Hamiltonian calculation
in cQED.

% --------------------------------------------------------------
\section{Time-dependent perturbation theory}
\label{sec:tdpt}

\begin{phenomenon}
A weakly perturbed system in an eigenstate of $\op{H}_0$ slowly
leaks probability amplitude into other eigenstates. To leading order
the leakage rate depends on a single matrix element of the
perturbation and on a frequency-domain mismatch between drive and
transition.
\end{phenomenon}

Let $\op{H}(t)=\op{H}_0+\op{V}(t)$ with $\op{H}_0\ket{n}=E_n\ket{n}$.
Expand the state in the interaction picture as
\begin{equation}
\ket{\psi(t)}_I = \sum_n c_n(t)\,\ket{n},
\end{equation}
where $\ket{\psi}_I=e^{i\op{H}_0 t/\hbar}\ket{\psi}_S$. Plugging into
the interaction-picture Schr\"odinger equation
$i\hbar\,\partial_t\ket{\psi}_I=\op{V}_I(t)\ket{\psi}_I$ and projecting
onto $\bra{f}$ gives
\begin{equation}
\label{eq:cdot-exact}
i\hbar\,\dot c_f(t) \;=\; \sum_n e^{i\omega_{fn}t}\,V_{fn}(t)\,c_n(t),
\qquad \omega_{fn}=(E_f-E_n)/\hbar,
\end{equation}
with $V_{fn}(t)=\bra{f}\op{V}(t)\ket{n}$.

\begin{statement}[First-order amplitude]
\label{stmt:first-order}
Starting from $\ket{i}$ at $t=0$ and assuming $\op{V}$ is small, the
amplitude to be in $\ket{f}\neq\ket{i}$ at time $t$ is
\begin{equation}
\label{eq:c-first-order}
c_f^{(1)}(t) \;=\; -\frac{i}{\hbar}\int_0^t e^{i\omega_{fi}t'}\,V_{fi}(t')\,\dd t'.
\end{equation}
\end{statement}

\begin{proof}
Set $c_n(t)=\delta_{ni}+c_n^{(1)}(t)+\ldots$ and substitute into
\cref{eq:cdot-exact}. Keeping only zeroth-order $c_n$ on the right
and $\dot c_f^{(1)}$ on the left gives
$i\hbar\,\dot c_f^{(1)}=e^{i\omega_{fi}t}V_{fi}(t)$; integrating from
$0$ to $t$ yields \cref{eq:c-first-order}.
\end{proof}

\paragraph{Transition.}
\Cref{eq:c-first-order} is exact at first order but contains only
amplitudes. To extract experimentally observable transition
\emph{rates} into a continuum of final states, we take the
long-time, dense-spectrum limit, which produces Fermi's golden rule.

% --------------------------------------------------------------
\section{Fermi's golden rule}
\label{sec:fermi}

\begin{theorem}[Fermi's golden rule]
\label{thm:fermi}
Let $\op{V}(t)=\op{V}\,e^{-i\omega t}+\op{V}^\dagger e^{+i\omega t}$ be a
harmonic perturbation, and let the final states form a continuum
with density of states $\rho_f(E)$ near $E=E_i+\hbar\omega$. The
transition rate from $\ket{i}$ to states near energy
$E_f=E_i+\hbar\omega$ is
\begin{equation}
\label{eq:fermi}
\Gamma_{i\to f} \;=\; \frac{2\pi}{\hbar}\,\bigl|\bra{f}\op{V}\ket{i}\bigr|^2\,\rho_f(E_f).
\end{equation}
\end{theorem}

\begin{proof}
From \cref{eq:c-first-order} with $V_{fi}(t)=V_{fi}\,e^{-i\omega t}$,
\begin{equation}
c_f^{(1)}(t) = -\frac{i}{\hbar}V_{fi}\,
\frac{e^{i(\omega_{fi}-\omega)t}-1}{i(\omega_{fi}-\omega)},
\end{equation}
so
\begin{equation}
\bigl|c_f^{(1)}(t)\bigr|^2
= \frac{|V_{fi}|^2}{\hbar^2}\,\frac{4\sin^2[(\omega_{fi}-\omega)t/2]}{(\omega_{fi}-\omega)^2}.
\end{equation}
For large $t$ the function on the right is sharply peaked at
$\omega_{fi}=\omega$ with integral $2\pi t$; in the distributional
sense $4\sin^2(xt/2)/x^2\to 2\pi t\,\delta(x)$. Summing over a
continuum of final states with density $\rho_f(E_f)$ and dividing by
$t$ gives \cref{eq:fermi}.
\end{proof}

\begin{example}[Spontaneous emission and the Purcell rate]
The decay rate of an excited atom into the continuum of vacuum
electromagnetic modes follows from Fermi's golden rule with
$|V_{fi}|^2$ proportional to the dipole matrix element squared and
$\rho_f(E_f)$ to the photon density of states. In a cavity,
modifying $\rho_f$ at the atomic frequency accelerates or suppresses
the decay --- the Purcell effect of \cref{sec:purcell}.
\end{example}

\paragraph{Transition.}
Fermi's golden rule applies to weak perturbations and incoherent
transitions. For \emph{strong} but \emph{near-resonant} drives, like
the microwave tones used to control qubits, a different
approximation is more useful: drop the fast off-resonant terms in
the interaction-picture Hamiltonian.

% --------------------------------------------------------------
\section{The rotating-wave approximation}
\label{sec:rwa}

\begin{phenomenon}
A near-resonant drive produces both a slow, ``co-rotating''
contribution that drives transitions coherently and a fast,
``counter-rotating'' one that, when integrated over time, makes
only a small bounded correction on top of the slow dynamics.
Provided the slow and fast scales are well separated, the
counter-rotating part can be dropped as a leading-order
approximation. The point is \emph{not} that the state ignores the
fast terms instantaneously --- it does not --- but that their
cumulative effect over the dynamical timescale stays bounded and
small.
\end{phenomenon}

\begin{statement}[RWA, generic form]
\label{stmt:rwa}
Consider $\op{H}(t)=\op{H}_0+\op{V}(t)$ with
$\op{H}_0\ket{n}=E_n\ket{n}$ and $\op{V}(t)$ harmonic at frequency
$\omega$. In the interaction picture,
$\op{V}_I(t)=\sum_{m,n} V_{mn}\,e^{i\omega_{mn}t}\,e^{\pm i\omega t}\,
\ket{m}\bra{n}$. Terms for which the total oscillation frequency
$|\omega_{mn}\pm\omega|$ is much larger than the interaction-strength
scale $|V_{mn}|/\hbar$ contribute, when integrated, only an
$O(|V_{mn}|/\hbar|\omega_{mn}\pm\omega|)$ correction to the state.
The RWA consists in dropping them and keeping only the resonant
terms with $\omega_{mn}\pm\omega\approx0$, which produce $O(1)$
dynamics on the timescale of interest.
\end{statement}

\paragraph{Why discarding the fast terms is legitimate.}
The naive ``fast oscillations average to zero'' justification is
not quite right: the Hamiltonian is non-zero at every instant and
the state \emph{is}, instantaneously, sensitive to every term in
it. The correct justification is about the \emph{time-integrated}
effect. From the first-order amplitude
\cref{eq:c-first-order}, the contribution of a single oscillating
piece $V_X\,e^{i\omega_X t}$ to the state at time $t$ is
proportional to
\begin{equation}
\label{eq:rwa-integral}
\int_0^t e^{i\omega_X t'}\,\dd t'
\;=\; \frac{e^{i\omega_X t}-1}{i\omega_X},
\qquad
\biggl|\int_0^t e^{i\omega_X t'}\,\dd t'\biggr| \;\leq\; \frac{2}{|\omega_X|}.
\end{equation}
The key observation is that this integral is \emph{bounded for all
$t$}, with a bound that shrinks as $|\omega_X|$ grows: the integral
does not accumulate with time, because each new oscillation
half-period cancels the previous one. A fast term therefore
generates an amplitude of at most $\sim|V_\text{fast}|/(\hbar\,\omega_\text{fast})$,
\emph{forever}.

A slow term, by contrast, has $\omega_X\to 0$, and the integral in
\cref{eq:rwa-integral} reduces to $t$: the amplitude grows
\emph{linearly} in time, reaching $O(1)$ on the dynamical timescale
$t\sim\hbar/|V_\text{slow}|$. Comparing the two, a fast term's
contribution is smaller than a slow term's, on that same
timescale, by the dimensionless ratio
\begin{equation}
\label{eq:rwa-small-parameter}
\epsilon_\text{RWA} \;\sim\; \frac{|V_\text{fast}|}{\hbar\,\omega_\text{fast}}.
\end{equation}
Whenever $\epsilon_\text{RWA}\ll 1$ --- i.e.\ whenever the matrix
element of the fast term is small compared to its oscillation
energy $\hbar\,\omega_\text{fast}$ --- the fast term acts as a
small additive correction on top of the slow dynamics. The RWA
consists in keeping the leading $O(1)$ behaviour and dropping the
$O(\epsilon_\text{RWA})$ correction. We are \emph{not} claiming
the fast terms have no effect; we are saying that their cumulative
effect on the dynamical timescale is bounded by
$\epsilon_\text{RWA}$ while the slow terms drive the leading
dynamics.

\begin{intuition}
Picture the state vector as a particle being pushed by a force that
contains a steady component plus a rapidly oscillating one. The
steady push moves the particle linearly in time. The oscillating
push moves it back and forth so quickly that the cumulative
displacement is at most one oscillation amplitude before the next
half-cycle reverses it. After many oscillation periods, the
steady displacement has grown without bound while the oscillating
contribution is still bounded by its single-amplitude maximum.
That is exactly the situation here, with ``displacement''
replaced by ``state-vector amplitude''.
\end{intuition}

The first systematic correction beyond the RWA is the
\emph{Bloch--Siegert shift}, an $O(\epsilon_\text{RWA}^2)$
correction to the effective qubit frequency. It enters at second
order because pairs of fast-term contributions can combine
coherently to leave a residual dc piece, and it becomes
experimentally measurable when the drive is so strong that
$\epsilon_\text{RWA}\sim0.1$ or larger.

\paragraph{Derivation, two-level example.}
For the driven TLS of \cref{sec:rabi} with
$\op{V}(t)=\hbar\Omega\cos(\omega_d t)\hat\sigma_x$, expand
$\hat\sigma_x=\hat\sigma_++\hat\sigma_-$ and write
\begin{equation}
\op{V}_I(t) = \tfrac{\hbar\Omega}{2}\bigl(\hat\sigma_+ e^{i\omega_q t}+\hat\sigma_- e^{-i\omega_q t}\bigr)
\bigl(e^{i\omega_d t}+e^{-i\omega_d t}\bigr).
\end{equation}
Multiplying out, four terms appear: two oscillate at
$\omega_q-\omega_d$ (slow, when $\omega_d\approx\omega_q$) and two
at $\pm(\omega_q+\omega_d)$ (always fast). Dropping the latter,
\begin{equation}
\op{V}_I^\text{RWA}(t)
= \tfrac{\hbar\Omega}{2}\bigl(\hat\sigma_+ e^{+i(\omega_q-\omega_d)t}+\hat\sigma_- e^{-i(\omega_q-\omega_d)t}\bigr).
\end{equation}
Going one step further into the frame rotating at the drive
frequency, $\op{U}_d(t)=e^{-i\omega_d t\hat\sigma_z/2}$, removes the
remaining time dependence and reproduces \cref{eq:rabi-h-rwa}, the
static Hamiltonian whose dynamics gave the Rabi formula.

\begin{remark}[Validity]
The RWA is exact only in the limit
$\Omega\ll \omega_q+\omega_d$ and $|\omega_q-\omega_d|\ll\omega_q+\omega_d$.
At very strong driving the kept terms produce dynamics on a timescale
comparable to the discarded fast oscillations, and corrections such
as the Bloch--Siegert shift become measurable.
\end{remark}

\begin{example}[Why we needed it earlier]
The Rabi formula in \cref{thm:rabi} was derived using exactly this
approximation. Every drive Hamiltonian we will meet from
\cref{ch:jaynes-cummings} onwards --- the Jaynes--Cummings coupling,
the dispersive readout drive, the cross-resonance two-qubit gate ---
takes its standard textbook form only after invoking the RWA.
\end{example}

This chapter completed the toolkit needed to handle time-dependent
problems: time-dependent perturbation theory gives transition
amplitudes order by order, Fermi's golden rule converts those
amplitudes into rates whenever a continuum of final states is
available, and the rotating-wave approximation simplifies
near-resonant drives to a static Hamiltonian in the appropriate
rotating frame.

Together with the postulates and operators of
\cref{ch:postulates}, the model systems of \cref{ch:model-systems},

formalism. From there Part~III turns to quantum optics proper, and
Part~IV to superconducting circuits, where every approximation
introduced in this chapter earns its keep.

\begin{chaptersummary}

\paragraph{Time-dependent perturbation theory.}
For $\op{H}=\op{H}_0+\op{V}(t)$ with $\op{V}$ small, expand the
state in the interaction picture as
$\ket{\psi}_I=\sum_n c_n(t)\ket{n}$. Starting from $\ket{i}$, the
first-order amplitude to be in $\ket{f}\neq\ket{i}$ at time $t$ is
\begin{equation}
c_f^{(1)}(t) \;=\; -\frac{i}{\hbar}\int_0^t e^{i\omega_{fi}t'}\,V_{fi}(t')\,\dd t',
\qquad \omega_{fi}=(E_f-E_i)/\hbar.
\end{equation}

\paragraph{Fermi's golden rule.}
For a harmonic perturbation
$\op{V}(t)=\op{V}\,e^{-i\omega t}+\op{V}^\dagger e^{+i\omega t}$ acting
into a continuum of final states with density of states
$\rho_f(E)$, the transition rate from $\ket{i}$ to states near
$E_f=E_i+\hbar\omega$ is
\begin{equation}
\Gamma_{i\to f} \;=\; \frac{2\pi}{\hbar}\,\bigl|\bra{f}\op{V}\ket{i}\bigr|^2\,\rho_f(E_f).
\end{equation}
This is the workhorse rate formula behind spontaneous emission, the
Purcell effect, and quasiparticle generation rates.

\paragraph{The rotating-wave approximation.}
Near-resonant drives produce both slow (co-rotating) and fast
(counter-rotating) terms. Although the Hamiltonian is non-zero at
every instant, the time-integrated effect of any oscillating piece
$V_X\,e^{i\omega_X t}$ is bounded:
\begin{equation}
\biggl|\int_0^t e^{i\omega_X t'}\,\dd t'\biggr| \;\leq\; \frac{2}{|\omega_X|}
\quad\text{for all }t.
\end{equation}
Slow contributions integrate to amplitudes that grow linearly in
time and become $O(1)$ on the dynamical timescale; fast contributions
remain bounded by a single oscillation amplitude forever. The
relevant small parameter is
\begin{equation}
\epsilon_\text{RWA} \;\sim\; \frac{|V_\text{fast}|}{\hbar\,\omega_\text{fast}},
\end{equation}
and the RWA consists in keeping the leading $O(1)$ slow dynamics
and dropping the $O(\epsilon_\text{RWA})$ fast contribution. The
first systematic correction beyond this is the Bloch--Siegert
shift, $O(\epsilon_\text{RWA}^2)$ on the qubit frequency.

\end{chaptersummary}

\section*{Exercises}
\addcontentsline{toc}{section}{Exercises}

\begin{exercise}[First-order amplitude for a constant perturbation]
Take $\op{V}(t)=\op{V}_0$ time-independent (suddenly switched on
at $t=0$) and compute $c_f^{(1)}(t)$ for $\ket{f}\neq\ket{i}$.
Comment on the behaviour at the resonance $E_f=E_i$.
\begin{solution}
\Cref{eq:c-first-order} gives
$c_f^{(1)}(t)=-(i/\hbar)V_{fi}\!\int_0^t\!e^{i\omega_{fi}t'}\dd t'
=-(i/\hbar)V_{fi}\,(e^{i\omega_{fi}t}-1)/(i\omega_{fi})
=-\,V_{fi}(e^{i\omega_{fi}t}-1)/(\hbar\omega_{fi})$.
$|c_f^{(1)}(t)|^2=4|V_{fi}|^2\sin^2(\omega_{fi}t/2)/(\hbar\omega_{fi})^2$
oscillates at frequency $\omega_{fi}$ with bounded amplitude.
At resonance ($\omega_{fi}\to 0$), L'H\^opital gives
$c_f^{(1)}(t)=-(i/\hbar)V_{fi}\,t$, growing linearly --- the
familiar first-order resonance signature that Fermi's golden rule
turns into a rate.
\end{solution}
\end{exercise}

\begin{exercise}[Sinc-squared limit and the delta function]
Verify the distributional limit
$4\sin^2(xt/2)/x^2 \xrightarrow{t\to\infty} 2\pi t\,\delta(x)$
that is needed in the derivation of \cref{thm:fermi}.
\begin{solution}
The function $f_t(x)=4\sin^2(xt/2)/x^2$ is non-negative, peaked
at $x=0$ with height $t^2$ and width $\sim 2\pi/t$, and
$\int_{-\infty}^\infty f_t(x)\dd x=2\pi t$ (a standard integral).
As $t\to\infty$ the peak narrows and grows; for any smooth test
function $g$, $\int f_t(x)g(x)\dd x\to 2\pi t\,g(0)$. By
definition this is $2\pi t\,\delta(x)$.
\end{solution}
\end{exercise}

\begin{exercise}[Cosine drive in the interaction picture: four exponentials]
Verify directly that
$\hbar\Omega\cos(\omega_d t)\hat\sigma_x$ in the interaction picture
generated by $\op{H}_0=\tfrac{\hbar\omega_q}{2}\hat\sigma_z$
produces four exponential terms; identify which are slow and
which are fast for $\omega_d\approx\omega_q$.
\begin{solution}
$\op{V}_I(t)=\op{U}_0^\dagger\hat\sigma_x\op{U}_0
\cdot\hbar\Omega\cos(\omega_d t)$, with
$\op{U}_0^\dagger\hat\sigma_x\op{U}_0=
e^{i\omega_q t}\hat\sigma_+ + e^{-i\omega_q t}\hat\sigma_-$
(since $\op{U}_0^\dagger\hat\sigma_\pm\op{U}_0=e^{\pm i\omega_q t}\hat\sigma_\pm$).
Multiplying by
$\cos(\omega_d t)=\tfrac12(e^{i\omega_d t}+e^{-i\omega_d t})$,
\[
\op{V}_I(t)=\tfrac{\hbar\Omega}{2}\bigl[
e^{i(\omega_q-\omega_d)t}\hat\sigma_+ + e^{-i(\omega_q-\omega_d)t}\hat\sigma_-
+ e^{i(\omega_q+\omega_d)t}\hat\sigma_+ + e^{-i(\omega_q+\omega_d)t}\hat\sigma_-\bigr].
\]
Slow ($\omega_d\approx\omega_q$): the first two with frequency
$\omega_q-\omega_d$. Fast: the last two with frequency
$\omega_q+\omega_d\approx 2\omega_q$.
\end{solution}
\end{exercise}

\begin{exercise}[Estimate $\epsilon_\text{RWA}$ for a transmon drive]
For a transmon at $\omega_q/2\pi=5\,\text{GHz}$ driven at
$\Omega/2\pi=50\,\text{MHz}$, estimate
$\epsilon_\text{RWA}=\Omega/(\omega_q+\omega_d)$ on resonance, and
the leading Bloch--Siegert frequency shift
$\Omega^2/(4\omega_q)$ on the qubit.
\begin{solution}
On resonance, $\omega_q+\omega_d=2\omega_q$, so
$\epsilon_\text{RWA}=\Omega/(2\omega_q)=50/(2\cdot 5000)=5\times 10^{-3}$.
The Bloch--Siegert shift to the rotating-frame qubit frequency is
$\Omega^2/(4\omega_q)/(2\pi)=(50)^2/(4\cdot 5000)\,\text{MHz}=0.125\,\text{MHz}=125\,\text{kHz}$,
a small but in principle measurable correction.
\end{solution}
\end{exercise}

\begin{exercise}[Time-dependent amplitude for an oscillating drive]
Take $V_{fi}(t)=V\cos(\omega t)$ and integrate
\cref{eq:c-first-order} for $\ket f\neq\ket i$. Identify the
co-rotating and counter-rotating contributions.
\begin{solution}
Write $\cos(\omega t)=\tfrac12(e^{i\omega t}+e^{-i\omega t})$, so
$c_f^{(1)}(t)=-(iV/2\hbar)\!\int_0^t\!\bigl(e^{i(\omega_{fi}+\omega)t'}+e^{i(\omega_{fi}-\omega)t'}\bigr)\dd t'
=-(V/2\hbar)\bigl[(e^{i(\omega_{fi}-\omega)t}-1)/(\omega_{fi}-\omega)+(e^{i(\omega_{fi}+\omega)t}-1)/(\omega_{fi}+\omega)\bigr]$.
The first bracket peaks near $\omega\approx\omega_{fi}$ (co-rotating,
absorption) and the second near $\omega\approx-\omega_{fi}$
(counter-rotating, ``emission''). Both terms appear naturally
because $\cos$ contains both frequencies.
\end{solution}
\end{exercise}

\begin{exercise}[Fermi's golden rule for a sinusoidal drive]
Using \cref{eq:c-first-order} with $V(t)=V\cos(\omega t)$ and the
sinc-squared limit, derive the absorption rate from $\ket i$ into
a continuum of final states with density $\rho_f(E_f)$. Compare
with \cref{thm:fermi}.
\begin{solution}
Only the resonant (co-rotating) term contributes at long times:
$|c_f^{(1)}(t)|^2\approx(V/2\hbar)^2\cdot 4\sin^2[(\omega_{fi}-\omega)t/2]/(\omega_{fi}-\omega)^2$.
Integrating over final states with density $\rho_f$ and using
$4\sin^2(xt/2)/x^2\to 2\pi t\,\delta(x)$,
$\sum_f|c_f^{(1)}|^2\to(V/2\hbar)^2 \cdot 2\pi t\,\rho_f(E_i+\hbar\omega)/\hbar
\cdot\hbar=(\pi V^2/2\hbar^2)\,\rho_f\,t$. Dividing by $t$ gives
the rate $\Gamma=(\pi V^2/2\hbar^2)\rho_f=
(2\pi/\hbar)|V/2|^2\rho_f$, which matches \cref{eq:fermi} with
$V\to V/2$ identified as the matrix element of the co-rotating
piece.
\end{solution}
\end{exercise}

\begin{exercise}[$\pi$-pulse duration and corrections]
A transmon at $\omega_q/2\pi=5\,\text{GHz}$ is driven on resonance
with $\Omega/2\pi=30\,\text{MHz}$. Compute the $\pi$-pulse duration
in the RWA. Estimate the relative error from neglecting
counter-rotating terms.
\begin{solution}
RWA: $t_\pi=\pi/\Omega=1/(2\cdot 30\,\text{MHz})\approx 16.7\,\text{ns}$.
Beyond-RWA correction is $O(\epsilon_\text{RWA})=\Omega/2\omega_q=
3\times 10^{-3}$, so the actual rotation angle deviates from $\pi$
by a few parts per thousand --- typically smaller than the
empirical pulse-amplitude calibration uncertainty, and absorbed
into the calibration.
\end{solution}
\end{exercise}

\begin{exercise}[Resonant absorption rate scales linearly in time at first order]
Show that for an exactly-resonant continuous drive in
\cref{eq:c-first-order}, $|c_f^{(1)}(t)|^2$ grows as $t^2$ at
first order, and explain why Fermi's golden rule gives a rate
(linear in $t$) only after integrating over a continuum of final
states.
\begin{solution}
At resonance, $\omega_{fi}=\omega$ and the integrand is constant,
so $c_f^{(1)}(t)=-(iV_{fi}/\hbar)t$, giving $|c|^2\propto t^2$.
This is not a rate. Summing over final states with density
$\rho_f$ and using the sinc-squared limit, only those final
states within $\sim 2\pi/t$ of resonance contribute coherently,
i.e.\ $\sim\rho_f\cdot 2\pi/t$ states each contributing $\sim
(V/\hbar)^2 t^2$. The product is $\sim 2\pi (V/\hbar)^2 \rho_f\,t$
--- linear in $t$, hence a finite rate. The $t^2$ amplitude growth
of an isolated resonance is a feature of coherent dynamics; the
linear-in-$t$ rate is what comes out after the destructive
interference among off-resonant final states is taken into account.
\end{solution}
\end{exercise}

\begin{exercise}[Validity of Fermi's golden rule]
On what time window is Fermi's golden rule a valid description of
the dynamics? What goes wrong at very short times, and at very
long times?
\begin{solution}
Validity requires (i) $t \gg 1/\Delta\omega_f$, where
$\Delta\omega_f$ is the spread in final-state frequencies, so that
the sinc$^2$ has narrowed onto resonance --- otherwise interference
between different final states has not yet washed out. And (ii)
$\Gamma t \ll 1$, so the perturbative expansion is still valid:
once the cumulative excitation probability becomes order unity the
first-order calculation breaks down and one needs to resum (e.g.\
master-equation methods, Part~II). Outside this window, at very
short times the dynamics is coherent (the $|c|^2\propto t^2$
behaviour above), and at very long times one sees saturation or
back-action that linear perturbation theory cannot capture.
\end{solution}
\end{exercise}

\begin{exercise}[A two-tone Rabi calculation]
Drive a qubit at frequency $\omega_q$ with two simultaneous tones,
$\op{V}(t)=\hbar\Omega_1\cos(\omega_1 t)\hat\sigma_x
+\hbar\Omega_2\cos(\omega_2 t)\hat\sigma_x$, with $\omega_1\approx\omega_q$
(near resonance) and $\omega_2$ far off resonance. Use the RWA to
write down the effective rotating-frame Hamiltonian. What is the
effective Rabi rate?
\begin{solution}
Move into the frame rotating at $\omega_1$. The first tone
contributes $\tfrac{\hbar\Omega_1}{2}\hat\sigma_x$ (slow, kept) and a
$\pm 2\omega_1$ counter-rotating part (fast, dropped). The second
tone, in the same frame, oscillates at
$\pm(\omega_2-\omega_1)$ and $\pm(\omega_2+\omega_1)$, all of which
are fast for $\omega_2$ far from $\omega_q$, and the leading
contribution is $O(\Omega_2/(\omega_2-\omega_1))$ smaller than the
first tone. To leading order the effective rotating-frame
Hamiltonian is
$\op{H}_R\approx\tfrac{\hbar(\omega_q-\omega_1)}{2}\hat\sigma_z
+\tfrac{\hbar\Omega_1}{2}\hat\sigma_x$, with effective Rabi rate
$\Omega_R=\sqrt{\Omega_1^2+(\omega_q-\omega_1)^2}$. The far-detuned
tone produces only a small AC-Stark-like correction, properly
treated in Part~III (\cref{sec:experimental-basics}).
\end{solution}
\end{exercise}
