\chapter{Postulates and the language of quantum mechanics}
\label{ch:postulates}


Quantum mechanics is the framework that replaces classical mechanics
when the systems we describe are small enough or cold enough that
their phase-space volume becomes comparable to $\hbar$. Before doing
any physics with it, we need a vocabulary: the Hilbert space in which
states live, the algebra of operators that represent observables, and
the rules that govern measurement and time evolution. This chapter
collects exactly that vocabulary. The development is concise and
deliberately abstract; intuition for what these objects mean will
be built in
\crefrange{ch:model-systems}{ch:perturbations} once we apply the
toolbox to concrete systems.

The chapter has five sections, plus one short reflective subsection
near its middle. \Cref{sec:states} introduces the Hilbert space of
states. \Cref{sec:operators} introduces operators, observables, and
the spectral theorem. \Cref{sec:measurement} states the Born rule
and the projection postulate; \cref{sec:quantum-randomness}, a
subsection of it, addresses three foundational questions about
quantum randomness that the formalism alone does not answer.
\Cref{sec:uncertainty} introduces commutators, the position and
momentum operators, and the Robertson uncertainty relation.
\Cref{sec:time-evolution} states the Schr\"odinger equation, the
Heisenberg and interaction pictures, and works through a driven
two-level system as a fully solved example. The chapter ends with
the usual summary block.

\section{States and Hilbert space}
\label{sec:states}

\begin{phenomenon}
Single-photon and single-electron interference experiments
demonstrate that quantum systems do not, in general, possess a
definite value of every observable prior to measurement. The full
list of possible outcomes for any measurement, together with the
probability of each, is encoded in a single abstract object that we
call the \emph{state} of the system.
\end{phenomenon}

\begin{postulate}[State space]
\label{post:states}
The state of an isolated quantum system is described by a unit
vector $\ket{\psi}\in\hilbert$ in a complex separable Hilbert space.
Two unit vectors that differ only by an overall phase, $\ket{\psi}$
and $e^{i\alpha}\ket{\psi}$, are \emph{physically equivalent}: no
measurement can distinguish them, because every observable
prediction depends only on $|\braket{\phi}{\psi}|^2$, in which the
global phase $e^{i\alpha}$ cancels (cf.\ the Born rule,
\cref{post:born}).
\end{postulate}

\begin{definition}[Hilbert space]
\label{def:hilbert}
A \emph{Hilbert space} $\hilbert$ is a complex vector space equipped
with an inner product
$\braket{\cdot}{\cdot}:\hilbert\times\hilbert\to\C$
(\cref{def:inner-product}: sesquilinear, conjugate-symmetric, and
positive definite; \cref{def:sesquilinear,def:conj-sym,def:positive-definite})
that is complete under the induced norm
$\|\psi\|=\sqrt{\braket{\psi}{\psi}}$ in the sense of
\cref{def:cauchy}. The Hilbert spaces relevant for these notes are
all \emph{separable} (\cref{def:separable}), i.e.\ they admit a
countable orthonormal basis.
\end{definition}

In Dirac (bra--ket) notation we write vectors as kets $\ket{\psi}$
and elements of the dual space (\cref{def:dual-space}) as bras
$\bra{\psi}$; the action of a bra on a ket is the inner product,
$\braket{\phi}{\psi}\in\C$. A countable orthonormal basis
$\{\ket{n}\}$ of $\hilbert$ (\cref{def:separable}) gives the
\emph{resolution of the identity}
\begin{equation}
\label{eq:resolution-identity}
\id \;=\; \sum_{n} \ket{n}\bra{n},
\end{equation}
which we will use ubiquitously to insert complete sets of states into
expressions.

\begin{example}[The simplest non-trivial Hilbert space]
The smallest Hilbert space that is not just $\C$ itself is
$\hilbert=\C^2$, with orthonormal basis $\{\ket{0},\ket{1}\}$.
Every unit vector in $\hilbert$ can be written, up to an overall
phase, as
\begin{equation}
\ket{\psi} = \cos(\theta/2)\,\ket{0} + e^{i\varphi}\sin(\theta/2)\,\ket{1},
\end{equation}
parametrised by a point $(\theta,\varphi)$ on the unit sphere $S^2$.
This is the Hilbert space of a two-level system (introduced as a
quantum-mechanical model in \cref{sec:tls}); the geometric picture
of the unit sphere will reappear there as the Bloch sphere
(\cref{sec:bloch}).
\end{example}

\paragraph{Transition.}
States by themselves are inert: they only become physical objects
once we say what we can \emph{measure} on them. Measurements are
mediated by operators, to which we now turn.

\section{Observables and operators}
\label{sec:operators}

\begin{phenomenon}
Some physical quantities --- energy, position, spin component along
a given axis --- can in principle be measured, while others cannot
be measured simultaneously without disturbance. Both facts must be
encoded in the mathematical object that represents a measurable
quantity.
\end{phenomenon}

\begin{postulate}[Observables]
\label{post:observables}
Every physical observable $A$ is represented by a self-adjoint
linear operator $\op{A}$ on $\hilbert$ (in finite dimensions
equivalently called Hermitian; in infinite dimensions self-adjoint
is the strictly correct condition --- see \cref{def:linear-op}).
The possible outcomes of a measurement of $A$ are the eigenvalues
of $\op{A}$.
\end{postulate}

\begin{definition}[Hermitian, unitary, projector]
\label{def:operator-types}
Let $\op{A}$ be a linear operator on $\hilbert$
(\cref{def:linear-op}), and $\op{A}^\dagger$ its adjoint
(\cref{def:adjoint}). We call $\op{A}$
\begin{itemize}
  \item \emph{Hermitian} (or self-adjoint) if $\op{A}^\dagger=\op{A}$;
  \item \emph{unitary} if $\op{A}^\dagger\op{A}=\op{A}\op{A}^\dagger=\id$;
  \item a \emph{projector} if $\op{A}^2=\op{A}$ and $\op{A}^\dagger=\op{A}$.
\end{itemize}
\end{definition}

\begin{theorem}[Spectral theorem, finite-dimensional version]
\label{thm:spectral}
Any Hermitian operator $\op{A}$ on a finite-dimensional Hilbert
space admits an orthonormal eigenbasis. Equivalently, there exist
real numbers $\{a_n\}$ and a complete set of orthogonal projectors
$\{\op{P}_n\}$ with $\op{P}_n\op{P}_m=\delta_{nm}\op{P}_n$ and
$\sum_n\op{P}_n=\id$ such that
\begin{equation}
\label{eq:spectral-decomp}
\op{A} \;=\; \sum_n a_n\,\op{P}_n.
\end{equation}
\end{theorem}

\begin{proof}[Sketch]
For finite-dimensional $\hilbert$ the proof reduces to the linear
algebra fact that any Hermitian matrix is unitarily diagonalisable.
For infinite-dimensional $\hilbert$ a more careful spectral measure
is required; we will not need that level of generality here.
\end{proof}

\begin{remark}
Unitaries play a dual role: as operators that conserve the inner
product,
$\braket{\hat{U}\phi}{\hat{U}\psi}=\braket{\phi}{\psi}$, and
therefore preserve probabilities; and as the operators that implement
reversible time evolution (\cref{sec:time-evolution}).
\end{remark}

\paragraph{Transition.}
Equipped with states and operators, we can now state how the two
combine to give an actual measurement outcome.

\section{Measurement and the Born rule}
\label{sec:measurement}

\begin{phenomenon}
A measurement of a quantum system in general yields one of several
possible outcomes, with probabilities determined by the state. After
the measurement, the system's state is updated to be consistent with
the outcome that was observed.
\end{phenomenon}

\begin{postulate}[Born rule and projective measurement]
\label{post:born}
Let $\ket{\psi}\in\hilbert$ and let $\op{A}=\sum_n a_n\op{P}_n$ be
an observable. A measurement of $\op{A}$ yields the outcome $a_n$
with probability
\begin{equation}
\label{eq:born}
p(a_n) \;=\; \bra{\psi}\op{P}_n\ket{\psi},
\end{equation}
and immediately after the measurement, provided $p(a_n)>0$, the
state is updated to
\begin{equation}
\label{eq:projection}
\ket{\psi}\;\longmapsto\; \frac{\op{P}_n\ket{\psi}}{\sqrt{p(a_n)}}.
\end{equation}
\end{postulate}

\begin{example}[Measuring $\hat\sigma_z$ on a two-level system]
Take $\op{A}=\hat\sigma_z$ with eigendecomposition
$\hat\sigma_z=(+1)\ket{0}\bra{0}+(-1)\ket{1}\bra{1}$. For
$\ket{\psi}=\cos(\theta/2)\ket{0}+e^{i\varphi}\sin(\theta/2)\ket{1}$
the Born rule gives $p(+1)=\cos^2(\theta/2)$ and
$p(-1)=\sin^2(\theta/2)$ for the eigenvalues $\pm 1$ of
$\hat\sigma_z$, independent of $\varphi$.
\end{example}

The expectation value of $\op{A}$ in state $\ket{\psi}$ is the
classical average of the outcomes weighted by their probabilities,
\begin{equation}
\label{eq:expectation}
\expval{\op{A}}_\psi \;=\; \sum_n a_n\,p(a_n)
\;=\; \bra{\psi}\op{A}\ket{\psi}.
\end{equation}

\subsection{On the nature of quantum randomness}
\label{sec:quantum-randomness}

The Born rule (\cref{post:born}) is unusual among the laws of
physics: it asserts an \emph{irreducible} probability distribution
over measurement outcomes. Three questions naturally follow that
the formalism by itself does not answer, but on which there is now
a substantial experimental and theoretical literature. We address
each briefly. The reader should know up front that none of the
answers affects the calculations done in the rest of these notes
--- every cQED experiment gives the same numerical predictions in
every interpretation --- but the discussion sets out what we are,
and what we are not, claiming when we apply the Born rule.

\paragraph{Bell's case for irreducible randomness.}
A natural worry is that the Born-rule probabilities reflect only
\emph{ignorance} of some hidden degree of freedom, in the way that
a classical coin toss looks random only because we do not track the
coin's exact initial conditions. If so, a more detailed
(``hidden-variable'') theory would in principle reproduce the
quantum predictions while restoring determinism.

The question of whether such a theory is possible was reduced to an
experimental one by Bell~\cite{bell1964}, who showed that any
\emph{local} hidden-variable theory --- one in which the underlying
variables associated with each measurement are determined by data
within the past light cone of that measurement --- must satisfy a
mathematical inequality that quantum mechanics, in suitable
correlation experiments, violates. Subsequent measurements have
confirmed quantum mechanics: the early experiments of Aspect,
Grangier and Roger~\cite{aspect1982} and, after several decades of
work to close experimental loopholes (detection efficiency,
measurement-setting independence, spacelike separation of
detectors), the simultaneous loophole-free Bell tests of
2015~\cite{hensen2015,giustina2015,shalm2015}.

\paragraph{The argument in CHSH form.}
The cleanest version of Bell's theorem is the one due to Clauser,
Horne, Shimony and Holt~\cite{chsh1969} (CHSH), which is the form
all of the experiments above are cast in. The setup involves two
distant observers who share a stream of entangled pairs (one of the
maximally entangled Bell states of \cref{ex:bell-states} is the
canonical choice). On each pair, Alice independently and randomly
chooses one of two binary measurement settings labelled
$\op{A}_1, \op{A}_2$, and Bob independently chooses one of two
$\op{B}_1, \op{B}_2$; each measurement returns an outcome in
$\{-1, +1\}$. Repeating many times yields the four correlation
coefficients
\begin{equation}
E_{ij} \;=\; \langle A_i\,B_j\rangle,
\qquad i,j \in \{1,2\},
\end{equation}
where the angled brackets are sample averages of the product of the
two outcomes. The \emph{CHSH quantity} is
\begin{equation}
\label{eq:chsh}
S \;=\; E_{11} + E_{12} + E_{21} - E_{22}.
\end{equation}

In any local hidden-variable theory, $S$ is bounded:
\begin{equation}
\label{eq:chsh-bound}
|S| \;\leq\; 2.
\end{equation}
The argument is short. At the level of a single event, the four
outcomes $A_1, A_2, B_1, B_2$ are all $\pm 1$ and are deterministic
functions of the hidden variables and the local settings. Then
\begin{equation}
A_1 B_1 + A_1 B_2 + A_2 B_1 - A_2 B_2
\;=\; A_1\,(B_1 + B_2) + A_2\,(B_1 - B_2)
\;\in\; \{-2,\, +2\},
\end{equation}
because $B_1, B_2 \in \{-1, +1\}$ forces exactly one of the
parentheses to be zero and the other to be $\pm 2$. Averaging both
sides over the joint distribution of hidden variables and settings
gives \cref{eq:chsh-bound}.

Quantum mechanics, applied to a maximally entangled pair with
measurement axes rotated by $22.5^\circ$ between adjacent pairs,
predicts
\begin{equation}
\label{eq:tsirelson}
S^{\text{QM}} \;=\; 2\sqrt 2 \;\approx\; 2.828
\end{equation}
(the so-called \emph{Tsirelson bound}, the largest value $|S|$ can
take in any quantum theory). The four loophole-free experiments
cited above measure $S$ in the range roughly $2.4$--$2.8$, well
above the classical bound $2$ and consistent with $S^{\text{QM}}$
within statistical error. The conclusion is that the world cannot
be both local \emph{and} described by hidden variables.

The supported conclusion is precise: \emph{no local hidden-variable
theory can reproduce the predictions of quantum mechanics}, under
the standard assumptions of locality and \emph{measurement
independence} (the experimenter's choice of setting is statistically
independent of the hidden variables). The Born-rule randomness is
in this sense irreducible within these two assumptions. This is
not the same as a metaphysical claim of ``true randomness'':
theories that abandon locality (such as the de Broglie--Bohm
pilot-wave theory~\cite{bohm1952}, which is explicitly non-local)
or measurement independence (the superdeterminism loophole below)
reproduce all known experiments while remaining deterministic. What
the Bell-inequality tests do show is that one cannot keep all of
locality, measurement independence, and hidden-variable determinism
simultaneously --- a constraint on what any future ``deeper''
theory can look like.

\paragraph{The superdeterminism loophole.}
A more radical way out of the Bell argument is to question one of
its silent assumptions: the \emph{statistical independence} between
the experimenter's choice of measurement setting and the hidden
variables of the entangled pair. The CHSH bound \cref{eq:chsh-bound}
is derived by averaging over hidden variables \emph{and} settings
\emph{independently}; if the two were correlated by a common cause
in their joint past, the bound need not hold. In its strongest
form, this loophole pictures every variable in the universe ---
including the random-number generators in Alice's and Bob's
laboratories --- as fixed by initial conditions at the Big Bang,
when all of today's spacelike-separated regions were within a
single light cone, and consistent in just the right way to
reproduce the apparent quantum correlations. Theories of this kind
are called \emph{superdeterministic}, and have been advocated, in
different forms, by 't~Hooft~\cite{thooft2016} and by Hossenfelder
and Palmer~\cite{hossenfelder2020}.

Superdeterminism is consistent with all known experiments,
including the loophole-free Bell tests, because no experiment can
guarantee that the hardware used to choose measurement settings is
uncorrelated with the underlying microphysics. Recent ``cosmic
Bell tests'', which seed the setting choices using fluctuations of
light from distant quasars emitted billions of years
ago~\cite{rauch2018}, push the required common-cause horizon
deeper into the past, but cannot rule out the loophole entirely.
From a pragmatic standpoint, superdeterminism is in tension with
the methodological assumptions of all of empirical science (which
takes for granted that experimenters can independently sample
inputs), but it remains a logically consistent way to recover
local determinism, and is mentioned here for completeness.

\paragraph{The fate of the wave function on measurement.}
A naive reading of the projection rule \cref{eq:projection} is that
the system \emph{really was} in the superposition $\sum_n c_n\ket{n}$
before the measurement and \emph{really is} in $\ket{n_0}$ after it
--- the rest of the wave function ``collapsing'' to nothing. Taken
literally, this is uncomfortable: the joint dynamics of system,
measurement apparatus and environment is, by every interaction
Hamiltonian we know how to write down, unitary, and unitary
evolution conserves information.

The standard resolution is that the projection postulate is an
\emph{effective} description, valid for an observer with access only
to the system after the system has interacted with an apparatus
that is itself entangled with a vast bath of degrees of freedom
(the readout electronics, the cryostat, the room, the cosmos). The
mathematical mechanism is the partial-trace construction set up in
\cref{sec:partial-trace} and made concrete by
\cref{ex:decoherence-by-partial-trace}, which carries out the same
analysis in a single line of algebra for one system qubit and one
environment qubit. The phase information of the system is not
destroyed; it is dispersed into system--apparatus--environment
correlations that we discard because we have no realistic way to
track them. From the system's point of view this dispersal is
indistinguishable from genuine non-unitary collapse; from the joint
system--environment point of view, the evolution is unitary
throughout. This view of measurement, developed systematically by
Zurek and collaborators under the name \emph{environment-induced
superselection} (or einselection)~\cite{zurek2003} and reviewed at
book length by Schlosshauer~\cite{schlosshauer2007}, is the
dominant practical perspective in quantum optics and cQED.

What the decoherence argument does \emph{not} explain is why we
observe a single outcome rather than a coherent superposition of
outcomes. The answer to that depends on the interpretation one
adopts.

\paragraph{The main interpretations.}
The mathematical content of quantum mechanics --- the Hilbert-space
formalism, unitary evolution, the Born rule --- is not in dispute.
What \emph{is} in dispute is the metaphysics: what (if anything)
the wave function corresponds to in reality, and what happens at a
measurement. The most widely discussed interpretations, and where
each places the ``extra'' structure beyond the bare formalism,
are:
\begin{itemize}
  \item \emph{Copenhagen} (Bohr, Heisenberg).
  The wave function describes the observer's knowledge, not an
  objective state. A ``classical/quantum cut'' is drawn between
  observer and observed; on the observer's side everything is
  classical, and the Born rule is an irreducible postulate. No
  further explanation of measurement is required.
  \item \emph{Many-worlds / relative-state} (Everett~\cite{everett1957}).
  The wave function is the only physical object and evolves only
  unitarily. ``Outcomes'' are decoherent branches of a universal
  wave function that no longer interfere; an observer perceives
  one branch but all branches are equally real. The Born
  probabilities must be derived from the branching structure
  itself, which is the subject of an ongoing technical literature.
  \item \emph{de Broglie--Bohm pilot wave}~\cite{bohm1952}.
  The wave function is a real, non-local guiding field for definite
  particle trajectories. Outcomes are deterministic given full
  initial conditions; the apparent randomness comes from
  initial-condition ignorance. The theory is empirically equivalent
  to standard quantum mechanics and explicitly non-local in the
  sense forced by Bell's theorem.
  \item \emph{QBism} (Fuchs--Mermin--Schack~\cite{fuchs2014}).
  The wave function is an agent's personal degree of belief about
  the outcomes of measurements they may perform. The Born rule is a
  normative coherence condition on those beliefs, and
  ``measurement'' is just a Bayesian update.
\end{itemize}
Several other consistent interpretations exist (consistent
histories, modal interpretations, spontaneous-collapse models such
as GRW, and so on); Cabello~\cite{cabello2017} provides a useful
map of the broader territory. Crucially, the interpretations listed
above are \emph{empirically equivalent} for every experiment we
will analyse in these notes: they are choices of conceptual framing,
not of physical content.

\paragraph{Practical takeaway.}
For doing physics on cQED hardware, the operationally relevant
story is short: \cref{eq:born} gives the probabilities of
measurement outcomes, and the apparent ``collapse'' is what
happens when the system becomes entangled with the readout chain
and the measurement record is recorded. Whether one then says the
unobserved branches have ``vanished'', ``become inaccessible'',
``always existed'', or ``were never real'' is a question the
apparatus does not answer. What the apparatus answers, very
precisely, is the histogram of outcomes and the post-measurement
state of the system. We use the Born rule and the partial trace to
compute both throughout the rest of these notes, and leave the
metaphysics to the interpretation of the reader's choice.

\section{Commutators and the uncertainty principle}
\label{sec:uncertainty}

\begin{definition}[Commutator and anticommutator]
For operators $\op{A},\op{B}$ define
\begin{equation}
[\op{A},\op{B}] = \op{A}\op{B}-\op{B}\op{A},
\qquad
\{\op{A},\op{B}\} = \op{A}\op{B}+\op{B}\op{A}.
\end{equation}
\end{definition}

In finite dimensions (and, with appropriate spectral assumptions,
in the infinite-dimensional cases relevant for these notes), two
Hermitian operators share an orthonormal eigenbasis if and only if
$[\op{A},\op{B}]=0$. When $[\op{A},\op{B}]\neq0$ the two observables
\emph{cannot be simultaneously diagonalised}, so they cannot be
assigned sharp values for every state at once --- although a
particular state may still happen to give a sharp value for both
when it lies in their joint eigenspace. The Robertson relation
below makes the obstruction quantitative.

\begin{theorem}[Robertson uncertainty relation]
\label{thm:robertson}
For any state $\ket{\psi}$ and Hermitian operators $\op{A},\op{B}$,
\begin{equation}
\label{eq:robertson}
\Delta A\,\Delta B \;\geq\; \tfrac{1}{2}\,
\bigl|\expval{[\op{A},\op{B}]}_\psi\bigr|,
\end{equation}
where $\Delta A=\sqrt{\expval{\op{A}^2}-\expval{\op{A}}^2}$.
\end{theorem}

\begin{proof}
Define the shifted operators
$\delta\op{A}=\op{A}-\expval{\op{A}}\,\id$ and similarly
$\delta\op{B}$, and the auxiliary state
$\ket{f}=\delta\op{A}\ket{\psi}$,
$\ket{g}=\delta\op{B}\ket{\psi}$. The Cauchy--Schwarz inequality
gives
\begin{equation}
\bigl|\braket{f}{g}\bigr|^2 \;\leq\; \braket{f}{f}\,\braket{g}{g}
\;=\; (\Delta A)^2(\Delta B)^2.
\end{equation}
Decomposing
$\braket{f}{g}=\expval{\delta\op{A}\,\delta\op{B}}=
\tfrac12\expval{\{\delta\op{A},\delta\op{B}\}}+\tfrac12\expval{[\op{A},\op{B}]}$,
the first term is real and the second imaginary, so
$|\braket{f}{g}|^2\geq\tfrac14|\expval{[\op{A},\op{B}]}|^2$.
Combining the two inequalities yields \cref{eq:robertson}.
\end{proof}

\begin{definition}[Position and momentum operators]
\label{def:position-momentum}
For a single non-relativistic particle moving in one spatial
dimension, the \emph{position} operator $\op{x}$ and the
\emph{momentum} operator $\op{p}$ are the self-adjoint operators on
the particle's Hilbert space defined, in the position
representation $\psi(x)\equiv\braket{x}{\psi}$, by
\begin{equation}
(\op{x}\,\psi)(x) \;=\; x\,\psi(x),
\qquad
(\op{p}\,\psi)(x) \;=\; -i\hbar\,\dv{x}\,\psi(x).
\end{equation}
Applying both sides of $\op{x}\op{p}-\op{p}\op{x}$ to a test wave
function $\psi(x)$ and using the product rule yields the
\emph{canonical commutation relation}
\begin{equation}
\label{eq:ccr}
[\op{x},\op{p}] \;=\; i\hbar\,\id.
\end{equation}
In several dimensions or for several particles, the same
construction is repeated for each independent coordinate, with
$[\op{x}_i,\op{p}_j]=i\hbar\,\delta_{ij}\,\id$ and all other pairs
commuting. We will need only \cref{eq:ccr} in what follows.
Strictly, $\op{x}$ and $\op{p}$ are unbounded operators with
distinct natural domains and the canonical commutation relation
holds on the dense subspace where both compositions
$\op{x}\op{p}\psi$ and $\op{p}\op{x}\psi$ make sense (see
\cref{def:linear-op} for the unbounded-operator caveat); for the
present chapter we assume sufficient smoothness of the wave
functions involved.
\end{definition}

\begin{example}[Position and momentum: the Heisenberg inequality]
Applying the Robertson relation \cref{eq:robertson} to
$\op{A}=\op{x}$, $\op{B}=\op{p}$ and using \cref{eq:ccr},
$|\expval{[\op{x},\op{p}]}|=\hbar$, one recovers the familiar
Heisenberg inequality
\begin{equation}
\Delta x\,\Delta p \;\geq\; \frac{\hbar}{2},
\end{equation}
saturated by Gaussian wave packets.
\end{example}

Static states and a probability rule are not yet a theory of motion.
We close the chapter by giving the equation that drives the state
forward in time.

\section{Time evolution}
\label{sec:time-evolution}

\begin{postulate}[Schr\"odinger equation]
\label{post:schrodinger}
The state of an isolated quantum system evolves continuously in
time according to
\begin{equation}
\label{eq:schrodinger}
i\hbar\,\dv{t}\ket{\psi(t)} \;=\; \op{H}(t)\,\ket{\psi(t)},
\end{equation}
where $\op{H}(t)$ is the Hermitian Hamiltonian operator.
\end{postulate}

For a time-independent Hamiltonian the formal solution is
$\ket{\psi(t)}=\op{U}(t)\ket{\psi(0)}$ with the unitary
$\op{U}(t)=e^{-i\op{H}t/\hbar}$. For time-dependent Hamiltonians
the solution requires the time-ordered exponential
\begin{equation}
\op{U}(t,t_0) = \mathcal{T}\exp\!\Bigl[-\tfrac{i}{\hbar}\!\int_{t_0}^t\!\op{H}(t')\,\dd t'\Bigr],
\end{equation}
which we will need in \cref{ch:perturbations}.

\subsection{Schr\"odinger, Heisenberg and interaction pictures}
\label{sec:pictures}

\paragraph{The freedom of choice.}
Every physical prediction is an expectation value of the form
$\bra{\psi(t)}\op{O}\ket{\psi(t)}$. Inserting any unitary
$\op{W}(t)$ in the middle ($\op{W}^\dagger\op{W}=\id$) leaves the
prediction unchanged:
\begin{equation}
\bra{\psi(t)}\op{O}\ket{\psi(t)}
\;=\; \bra{\psi(t)}\op{W}^\dagger\,\bigl(\op{W}\op{O}\op{W}^\dagger\bigr)\,\op{W}\ket{\psi(t)}.
\end{equation}
By choosing $\op{W}$ we can shift time dependence between the state
and the operator without changing any measurable quantity. The
three standard choices are:
\begin{itemize}
  \item \emph{Schr\"odinger picture}: $\op{W}=\id$. States carry
        all the time dependence ($\ket{\psi(t)}_S$); operators are
        time-independent ($\op{O}_S$). This is the picture in
        which we stated the Schr\"odinger equation
        \cref{eq:schrodinger}.
  \item \emph{Heisenberg picture}: $\op{W}=\op{U}^\dagger(t)$ with
        $\op{U}(t)=e^{-i\op{H}t/\hbar}$. The unitary
        $\op{W}=\op{U}^\dagger$ exactly undoes the Schr\"odinger
        evolution of the state, so states become time-independent
        and operators inherit the dynamics.
  \item \emph{Interaction picture}: $\op{W}=\op{U}_0^\dagger(t)$
        with $\op{U}_0(t)=e^{-i\op{H}_0 t/\hbar}$ for some
        \emph{reference} Hamiltonian $\op{H}_0$ (a piece of the
        full $\op{H}=\op{H}_0+\op{V}(t)$ that we want to ``rotate
        out''). States and operators both evolve, but only under
        the parts of $\op{H}$ that the reference does not capture.
\end{itemize}

\begin{statement}[Picture transformations]
\label{stmt:pictures}
Let $\op{H}=\op{H}_0+\op{V}(t)$ with $\op{H}_0$
\emph{time-independent}, and let $\op{U}(t)$ be the propagator
generated by $\op{H}$ and $\op{U}_0(t)=e^{-i\op{H}_0 t/\hbar}$ the
propagator generated by $\op{H}_0$. For time-independent $\op{H}$
the full propagator reduces to $\op{U}(t)=e^{-i\op{H}t/\hbar}$; for
time-dependent $\op{H}$ it is the time-ordered exponential
\begin{equation}
\label{eq:full-prop-T-ordered}
\op{U}(t,t_0) \;=\; \mathcal{T}\!\exp\!\Bigl[-\tfrac{i}{\hbar}\!\int_{t_0}^t \op{H}(t')\,\dd t'\Bigr],
\end{equation}
which equals the simple exponential only when $\op{H}(t)$ commutes
with itself at different times. Schr\"odinger states/operators are
denoted with subscript $S$, Heisenberg with $H$, interaction with
$I$. Then
\begin{align}
\ket{\psi}_H &\;\equiv\; \op{U}^\dagger(t)\,\ket{\psi(t)}_S
              \;=\; \ket{\psi(0)}_S
\quad(\text{time-independent}), &
\op{O}_H(t) &\;\equiv\; \op{U}^\dagger(t)\,\op{O}_S(t)\,\op{U}(t),
\\[2pt]
\ket{\psi(t)}_I &\;\equiv\; \op{U}_0^\dagger(t)\,\ket{\psi(t)}_S, &
\op{O}_I(t) &\;\equiv\; \op{U}_0^\dagger(t)\,\op{O}_S(t)\,\op{U}_0(t).
\end{align}
The dynamical equations in each picture are
\begin{equation}
\dv{t}\op{O}_H \;=\; \op{U}^\dagger(t)\,\bigl(\partial_t\op{O}_S\bigr)\,\op{U}(t)
\,+\, \tfrac{i}{\hbar}[\op{H}_H(t),\op{O}_H(t)],
\qquad
i\hbar\,\dv{t}\ket{\psi(t)}_I \;=\; \op{V}_I(t)\,\ket{\psi(t)}_I,
\end{equation}
where $\op{H}_H(t)=\op{U}^\dagger(t)\op{H}(t)\op{U}(t)$. The first
term in the Heisenberg equation vanishes whenever $\op{O}_S$ has no
explicit time dependence, recovering the textbook form
$\dv{t}\op{O}_H=(i/\hbar)[\op{H},\op{O}_H]$.
\end{statement}

\begin{remark}[Why the Heisenberg state is constant in time]
The second equality on the Heisenberg line,
$\ket{\psi}_H=\ket{\psi(0)}_S$, is not an additional assumption: it
follows from the Schr\"odinger evolution
$\ket{\psi(t)}_S=\op{U}(t)\ket{\psi(0)}_S$. Substituting into the
definition,
\begin{equation}
\ket{\psi}_H \;=\; \op{U}^\dagger(t)\,\ket{\psi(t)}_S
              \;=\; \op{U}^\dagger(t)\,\op{U}(t)\,\ket{\psi(0)}_S
              \;=\; \ket{\psi(0)}_S.
\end{equation}
The defining unitary $\op{U}^\dagger(t)$ is, by construction, the
inverse of the Schr\"odinger flow, so applying it to
$\ket{\psi(t)}_S$ ``rewinds'' the state back to $t=0$. That is the
sense in which Heisenberg states do not move: all the dynamics has
been absorbed into the operators by the conjugation
$\op{O}_H(t)=\op{U}^\dagger\op{O}_S\op{U}$.
\end{remark}

\begin{proof}[Derivation, interaction picture]
Differentiating $\ket{\psi}_I=\op{U}_0^\dagger\ket{\psi}_S$,
\begin{equation}
i\hbar\,\dv{t}\ket{\psi}_I
= -\op{U}_0^\dagger\op{H}_0\ket{\psi}_S
+ \op{U}_0^\dagger\op{H}\ket{\psi}_S
= \op{U}_0^\dagger\,\op{V}\,\op{U}_0\,\ket{\psi}_I
= \op{V}_I(t)\,\ket{\psi}_I.
\end{equation}
The Heisenberg equation follows analogously by differentiating
$\op{O}_H=\op{U}^\dagger\op{O}_S\op{U}$ and using
$i\hbar\,\dot{\op{U}}=\op{H}\op{U}$.
\end{proof}

\begin{example}[Two-level system in a rotating frame]
\label{ex:tls-rotating-frame}
The interaction picture is the workhorse tool for ``rotating away''
a known piece of the Hamiltonian. To see it in action we work
through the simplest non-trivial case: a system with two energy
eigenstates $\ket{0}$ and $\ket{1}$ (energies $0$ and $\hbar\omega_0$,
respectively) coupled by a weak periodic drive.

\paragraph{Setup.}
Take
\begin{equation}
\op{H}_0 \;=\; \hbar\omega_0\,\ket{1}\bra{1},
\qquad
\op{V}(t) \;=\; \hbar\Omega\cos(\omega_d t)\,
\bigl(\ket{0}\bra{1}+\ket{1}\bra{0}\bigr),
\end{equation}
with a small drive amplitude $\Omega>0$ and a drive frequency
$\omega_d$ that we will eventually pick close to $\omega_0$. The
operator $\ket{0}\bra{1}+\ket{1}\bra{0}$ swaps the two basis states,
so $\op{V}(t)$ is the off-diagonal coupling that lets the drive move
amplitude between $\ket{0}$ and $\ket{1}$.

\paragraph{Step 1: build the reference unitary.}
The reference Hamiltonian $\op{H}_0$ has eigenstates $\ket{0}$ and
$\ket{1}$, so
\begin{equation}
\op{U}_0(t) \;=\; e^{-i\op{H}_0 t/\hbar}
\;=\; \ket{0}\bra{0} + e^{-i\omega_0 t}\,\ket{1}\bra{1}.
\end{equation}
Conjugating $\ket{1}\bra{0}$ by $\op{U}_0$ produces a single
exponential factor:
\begin{equation}
\op{U}_0^\dagger(t)\,\ket{1}\bra{0}\,\op{U}_0(t)
\;=\; e^{+i\omega_0 t}\,\ket{1}\bra{0},
\quad\text{and}\quad
\op{U}_0^\dagger(t)\,\ket{0}\bra{1}\,\op{U}_0(t)
\;=\; e^{-i\omega_0 t}\,\ket{0}\bra{1}.
\end{equation}

\paragraph{Step 2: write $\op{V}_I(t)$.}
By definition $\op{V}_I(t)=\op{U}_0^\dagger(t)\,\op{V}(t)\,\op{U}_0(t)$,
so
\begin{equation}
\op{V}_I(t) \;=\; \hbar\Omega\cos(\omega_d t)\,
\bigl(e^{+i\omega_0 t}\ket{1}\bra{0} + e^{-i\omega_0 t}\ket{0}\bra{1}\bigr).
\end{equation}
Now expand the cosine, $\cos(\omega_d t)=\tfrac12(e^{i\omega_d t}+e^{-i\omega_d t})$,
and multiply out. Four exponential terms appear:
\begin{equation}
\op{V}_I(t) \;=\; \tfrac{\hbar\Omega}{2}\Bigl[\,
\underbrace{e^{i(\omega_0-\omega_d)t}\ket{1}\bra{0}
       + e^{-i(\omega_0-\omega_d)t}\ket{0}\bra{1}}_{\text{slow at $\omega_d\approx\omega_0$}}
\;+\;
\underbrace{e^{i(\omega_0+\omega_d)t}\ket{1}\bra{0}
       + e^{-i(\omega_0+\omega_d)t}\ket{0}\bra{1}}_{\text{always fast}}
\,\Bigr].
\end{equation}

\paragraph{Step 3: read off the physics.}
The first two terms oscillate at the \emph{detuning}
$\omega_0-\omega_d$, which is small whenever the drive is tuned
close to the level spacing $\omega_0$. They drive amplitude back
and forth between $\ket{0}$ and $\ket{1}$ slowly, and accumulate
coherently. The last two terms oscillate at $\omega_0+\omega_d
\approx 2\omega_0$ regardless of how the drive is tuned; they
average down to a small residual when integrated over many cycles.
Provided $\Omega\ll\omega_0+\omega_d$ (so that the slow dynamics is
itself slow compared to $2\omega_0$), the fast pair can be dropped.
Doing so is the \emph{rotating-wave approximation}, derived
systematically in \cref{sec:rwa}; the surviving slow Hamiltonian is
the starting point of the Rabi formula (\cref{sec:rabi}). We will
recognise these same four terms, written in the Pauli-operator
language of \cref{ch:model-systems}, when we revisit this
calculation there.
\end{example}

We now have everything we need to do physics: states are unit
vectors in a Hilbert space; observables are Hermitian operators;
measurements yield eigenvalues with probabilities given by the
Born rule; non-commuting observables obey the Robertson uncertainty
relation; and unitary evolution under the Schr\"odinger equation
governs the dynamics. The Heisenberg and interaction pictures give
complementary, frame-dependent descriptions of the same physics,
and the foundational excursion of \cref{sec:quantum-randomness}
placed the irreducible Born-rule randomness in its modern
experimental and interpretational context.

Equipped with this vocabulary, we now apply it to two systems
that, between them, account for the bulk of cQED dynamics: the
quantum harmonic oscillator (resonators; the leading behaviour of
the transmon) and the two-level system (qubits; the geometric
Bloch-sphere picture for single-qubit gates).

\clearpage
\begin{chaptersummary}

\paragraph{States.}
The state of an isolated quantum system is a unit vector
$\ket{\psi}\in\hilbert$ in a complex separable Hilbert space, defined
up to an overall phase $e^{i\alpha}$. The bra-ket inner product
$\braket{\phi}{\psi}\in\C$ is sesquilinear, conjugate-symmetric, and
positive definite (\cref{def:hilbert}); a countable orthonormal
basis $\{\ket{n}\}$ gives the resolution of the identity
$\id=\sum_n\ket{n}\bra{n}$.

\paragraph{Observables.}
Observables are Hermitian operators (\cref{def:operator-types}). The
spectral theorem decomposes any Hermitian
$\op{A}=\sum_n a_n\,\op{P}_n$, with the eigenvalues $a_n$ the
possible measurement outcomes and the projectors $\op{P}_n$
orthogonal and complete.

\paragraph{Measurement (Born rule).}
A measurement of $\op{A}$ on $\ket{\psi}$ yields outcome $a_n$ with
probability
\begin{equation}
p(a_n) \;=\; \bra{\psi}\op{P}_n\ket{\psi},
\qquad
\expval{\op{A}}_\psi \;=\; \bra{\psi}\op{A}\ket{\psi},
\end{equation}
and the post-measurement state is
$\op{P}_n\ket{\psi}/\sqrt{p(a_n)}$.

\paragraph{Commutators and uncertainty.}
The position and momentum operators satisfy
$[\op{x},\op{p}]=i\hbar\,\id$ (\cref{def:position-momentum}). The
Robertson relation
\begin{equation}
\Delta A\,\Delta B \;\geq\; \tfrac{1}{2}\,
\bigl|\expval{[\op{A},\op{B}]}_\psi\bigr|
\end{equation}
quantifies the cost of failing to commute, and reduces to the
Heisenberg inequality $\Delta x\,\Delta p\geq\hbar/2$ for $\op{x},\op{p}$.

\paragraph{Time evolution.}
The Schr\"odinger equation
\begin{equation}
i\hbar\,\dv{t}\ket{\psi(t)} \;=\; \op{H}(t)\,\ket{\psi(t)}
\end{equation}
generates the dynamics; for time-independent $\op{H}$, the
solution is $\ket{\psi(t)}=\op{U}(t)\ket{\psi(0)}$ with
$\op{U}(t)=e^{-i\op{H}t/\hbar}$.

\paragraph{Pictures.}
The same physics can be described in three frames:
\begin{itemize}
  \item Schr\"odinger: $\ket{\psi}_S$ evolves; operators do not.
  \item Heisenberg: states constant, operators evolve as
        $\op{O}_H(t)=\op{U}^\dagger(t)\,\op{O}_S\,\op{U}(t)$, with
        $\dv{t}\op{O}_H=\tfrac{i}{\hbar}[\op{H},\op{O}_H]$.
  \item Interaction: with respect to a chosen reference $\op{H}_0$,
        $\ket{\psi(t)}_I$ obeys
        $i\hbar\,\dv{t}\ket{\psi(t)}_I=\op{V}_I(t)\,\ket{\psi(t)}_I$
        where $\op{V}_I=\op{U}_0^\dagger\op{V}\op{U}_0$.
\end{itemize}

\paragraph{Foundations.}
Born-rule randomness is irreducible \emph{within locality}: no
local hidden-variable theory reproduces QM (Bell + the loophole-free
2015 experiments). Apparent measurement collapse is the partial
trace of system + apparatus + environment over apparatus and
environment, made concrete in \cref{ex:decoherence-by-partial-trace}.
The mainstream interpretations of QM (Copenhagen, many-worlds,
Bohmian, QBism) are empirically equivalent for cQED.

\end{chaptersummary}

\section*{Exercises}
\addcontentsline{toc}{section}{Exercises}

The exercises grow in difficulty from inspection-level checks of
the formalism to applied calculations of the kind one runs in a
qubit lab. Each exercise is followed by a worked solution.

\begin{exercise}[Pauli operators are Hermitian, unitary, square to identity]
Show, by direct matrix calculation, that the Pauli operators
$\hat\sigma_x,\hat\sigma_y,\hat\sigma_z$ are simultaneously
Hermitian and unitary, and that each squares to $\id$.
\begin{solution}
By inspection of \cref{def:pauli}, each of the three matrices is
equal to its conjugate transpose, so
$\hat\sigma_i^\dagger=\hat\sigma_i$ (Hermitian). Squaring,
$\hat\sigma_x^2=\bigl(\begin{smallmatrix}0&1\\1&0\end{smallmatrix}\bigr)^2
=\bigl(\begin{smallmatrix}1&0\\0&1\end{smallmatrix}\bigr)=\id$,
and the same calculation for $\hat\sigma_y$ and $\hat\sigma_z$.
Hence $\hat\sigma_i^\dagger \hat\sigma_i=\hat\sigma_i^2=\id$, so
each is unitary as well.
\end{solution}
\end{exercise}

\begin{exercise}[Probabilities on the Bloch sphere]
For $\ket{\psi}=\cos(\theta/2)\,\ket{0}+e^{i\varphi}\sin(\theta/2)\,\ket{1}$,
compute the probability of obtaining the $+1$ outcome for each of
$\hat\sigma_x,\hat\sigma_y,\hat\sigma_z$, and verify that the
relation $\langle\hat\sigma_i\rangle=2\,p_i(+1)-1$ recovers the
components of the Bloch vector
$\bm{r}=(\sin\theta\cos\varphi,\sin\theta\sin\varphi,\cos\theta)$.
\begin{solution}
The $+1$-eigenstates are $\ket{+_z}=\ket{0}$,
$\ket{+_x}=(\ket{0}+\ket{1})/\sqrt 2$,
$\ket{+_y}=(\ket{0}+i\ket{1})/\sqrt 2$. The Born rule yields
$p_z(+1)=|\bra{0}\ket{\psi}|^2=\cos^2(\theta/2)
=\tfrac12(1+\cos\theta)$, and similarly
$p_x(+1)=\tfrac12+\tfrac12\sin\theta\cos\varphi$,
$p_y(+1)=\tfrac12+\tfrac12\sin\theta\sin\varphi$.
Then $\langle\hat\sigma_z\rangle=2p_z(+1)-1=\cos\theta$,
$\langle\hat\sigma_x\rangle=\sin\theta\cos\varphi$,
$\langle\hat\sigma_y\rangle=\sin\theta\sin\varphi$, the Bloch
vector.
\end{solution}
\end{exercise}

\begin{exercise}[Pauli commutator]
Verify the algebra $[\hat\sigma_i,\hat\sigma_j]=2i\,\epsilon_{ijk}\hat\sigma_k$
for the case $i=x,\,j=y$.
\begin{solution}
Direct multiplication:
$\hat\sigma_x\hat\sigma_y=
\bigl(\begin{smallmatrix}0&1\\1&0\end{smallmatrix}\bigr)
\bigl(\begin{smallmatrix}0&-i\\i&0\end{smallmatrix}\bigr)
=\bigl(\begin{smallmatrix}i&0\\0&-i\end{smallmatrix}\bigr)
=i\hat\sigma_z$.
Similarly $\hat\sigma_y\hat\sigma_x=-i\hat\sigma_z$. Hence
$[\hat\sigma_x,\hat\sigma_y]=2i\hat\sigma_z=2i\,\epsilon_{xyz}\hat\sigma_z$,
matching the general formula with $\epsilon_{xyz}=+1$.
\end{solution}
\end{exercise}

\begin{exercise}[Robertson for $\hat\sigma_x,\hat\sigma_y$ in $\ket{0}$]
Apply the Robertson uncertainty relation \cref{eq:robertson} with
$\op{A}=\hat\sigma_x$ and $\op{B}=\hat\sigma_y$ in the state
$\ket{0}$. What is $\Delta\sigma_x\,\Delta\sigma_y$ and what is
the right-hand side bound?
\begin{solution}
For $\ket{0}$, $\langle\hat\sigma_x\rangle=\langle\hat\sigma_y\rangle=0$
and $\langle\hat\sigma_x^2\rangle=\langle\hat\sigma_y^2\rangle=1$
(since each Pauli squares to $\id$), so
$\Delta\sigma_x=\Delta\sigma_y=1$ and $\Delta\sigma_x\,\Delta\sigma_y=1$.
The right-hand side is
$\tfrac12|\langle[\hat\sigma_x,\hat\sigma_y]\rangle_0|=\tfrac12|2i\langle\hat\sigma_z\rangle_0|
=|\langle\hat\sigma_z\rangle_0|=1$. The inequality is saturated.
\end{solution}
\end{exercise}

\begin{exercise}[Canonical commutator on a wave function]
Verify $[\op{x},\op{p}]\,\psi(x)=i\hbar\,\psi(x)$ for an arbitrary
smooth wave function by computing both terms explicitly using
the position-representation forms of \cref{def:position-momentum}.
\begin{solution}
$(\op{x}\,\op{p}\,\psi)(x)=\op{x}\,(-i\hbar\,\psi'(x))=-i\hbar\,x\,\psi'(x)$,
and
$(\op{p}\,\op{x}\,\psi)(x)=-i\hbar\,\partial_x[x\,\psi(x)]
=-i\hbar\,\psi(x)-i\hbar\,x\,\psi'(x)$.
Subtracting,
$[\op{x},\op{p}]\,\psi(x)=-i\hbar\,x\,\psi'(x)+i\hbar\,\psi(x)+i\hbar\,x\,\psi'(x)
=i\hbar\,\psi(x)$, as required.
\end{solution}
\end{exercise}

\begin{exercise}[Time evolution under $\hat\sigma_z$ Hamiltonian]
For $\op{H}=(\hbar\omega/2)\hat\sigma_z$, compute the time-evolution
operator $\op{U}(t)=e^{-i\op{H}t/\hbar}$ as a $2\times 2$ matrix,
and apply it to $\ket{+}=(\ket{0}+\ket{1})/\sqrt 2$. Sketch the
Bloch-sphere trajectory.
\begin{solution}
Since $\hat\sigma_z^2=\id$,
$\op{U}(t)=\cos(\omega t/2)\id-i\sin(\omega t/2)\hat\sigma_z
=\bigl(\begin{smallmatrix}e^{-i\omega t/2}&0\\0&e^{+i\omega t/2}\end{smallmatrix}\bigr)$.
Acting on $\ket{+}$,
$\op{U}(t)\ket{+}=(e^{-i\omega t/2}\ket{0}+e^{+i\omega t/2}\ket{1})/\sqrt 2$.
Up to global phase, this equals
$(\ket{0}+e^{+i\omega t}\ket{1})/\sqrt 2$, with Bloch vector
$(\cos\omega t,\sin\omega t,0)$. The state precesses on the equator
of the Bloch sphere about the $z$-axis at angular speed $\omega$.
\end{solution}
\end{exercise}

\begin{exercise}[Heisenberg-conserved observables]
Show that, for time-independent $\op{H}$, the Heisenberg-picture
operator $\op{O}_H(t)$ is time-independent if and only if
$[\op{H},\op{O}]=0$. Conclude that
$\expval{\op{O}}_\psi(t)$ is conserved \emph{for every initial
state $\ket{\psi}$} if and only if $\op{O}$ commutes with $\op{H}$.
(For a particular initial state, $\expval{\op{O}}(t)$ may also
remain constant when $[\op{H},\op{O}]\neq0$ --- e.g.\ when
$\ket{\psi(0)}$ lies in a joint eigenspace of $\op{H}$ and
$\op{O}$.)
\begin{solution}
The Heisenberg equation gives
$\dv{t}\op{O}_H=\tfrac{i}{\hbar}[\op{H},\op{O}_H]$. The right-hand
side vanishes for all $t$ iff $[\op{H},\op{O}_H]=0$, i.e.\ iff
$[\op{H},\op{O}]=0$ (commutativity is preserved by conjugation by
$\op{U}$). For expectation values,
$\expval{\op{O}}_\psi(t)=\bra{\psi(0)}\op{O}_H(t)\ket{\psi(0)}$ in
Heisenberg, which is constant in $t$ \emph{for every $\ket{\psi(0)}$}
iff $\op{O}_H(t)$ is the same operator for every $t$, iff
$[\op{H},\op{O}]=0$. The state-by-state caveat is the standard
escape clause: a state in the joint eigenspace of $\op{H}$ and
$\op{O}$ has constant expectation values for both regardless of
the commutator.
\end{solution}
\end{exercise}

\begin{exercise}[Gaussian wave packet saturates Heisenberg]
For a Gaussian wave function
$\psi(x)=(\pi a^2)^{-1/4}\,e^{-x^2/(2a^2)}$ with $a>0$, compute
$\Delta x$ and $\Delta p$ and verify the Heisenberg inequality is
saturated: $\Delta x\,\Delta p=\hbar/2$.
\begin{solution}
By symmetry $\langle\op{x}\rangle=\langle\op{p}\rangle=0$.
Direct integration gives
$\langle\op{x}^2\rangle=\int x^2|\psi|^2\,\dd x=a^2/2$, so
$\Delta x=a/\sqrt 2$. For momentum, switch to Fourier space:
$\tilde\psi(p)\propto e^{-p^2 a^2/(2\hbar^2)}$, a Gaussian with
width $\hbar/a$ in $p$, so
$\langle\op{p}^2\rangle=\hbar^2/(2a^2)$ and
$\Delta p=\hbar/(a\sqrt 2)$. Hence
$\Delta x\,\Delta p=(a/\sqrt 2)(\hbar/(a\sqrt 2))=\hbar/2$,
saturating the Heisenberg inequality.
\end{solution}
\end{exercise}

\begin{exercise}[Pauli operators in the Heisenberg picture]
Under $\op{H}=(\hbar\omega/2)\hat\sigma_z$, compute the
Heisenberg-picture operators
$\hat\sigma_x(t)$ and $\hat\sigma_y(t)$ explicitly. Interpret the
result as a rotation in the $x$--$y$ plane.
\begin{solution}
Apply the formula
$\hat\sigma_x(t)=\op{U}^\dagger\hat\sigma_x\op{U}$ with
$\op{U}=e^{-i\omega t\hat\sigma_z/2}$ and using
$e^{-i\theta\hat\sigma_z/2}\hat\sigma_x e^{+i\theta\hat\sigma_z/2}
=\hat\sigma_x\cos\theta+\hat\sigma_y\sin\theta$ (which one verifies
from the BCH expansion or by direct multiplication),
\begin{align*}
\hat\sigma_x(t) &= \cos(\omega t)\,\hat\sigma_x - \sin(\omega t)\,\hat\sigma_y, \\
\hat\sigma_y(t) &= \cos(\omega t)\,\hat\sigma_y + \sin(\omega t)\,\hat\sigma_x.
\end{align*}
The two operators rotate rigidly in the $x$--$y$ plane at angular
speed $\omega$, mirroring the Larmor precession of the Bloch vector.
\end{solution}
\end{exercise}

\begin{exercise}[Thermal occupation of a 5\,GHz qubit at 50\,mK]
Estimate the equilibrium population of $\ket{1}$ for a qubit at
$\omega_q/2\pi=5\,\text{GHz}$ in equilibrium with a thermal bath at
$T=50\,\text{mK}$. Comment on the implication for state preparation
in cQED experiments.
\begin{solution}
At thermal equilibrium $p_1/p_0=e^{-\hbar\omega_q/k_B T}$. With
$\omega_q=2\pi f$, the energy quantum is $\hbar\omega_q=h f$, so
\[
\hbar\omega_q \;=\; h f
\;=\; (6.626\times 10^{-34}\,\text{J s})(5\times 10^9\,\text{Hz})
\;\approx\; 3.31\times 10^{-24}\,\text{J},
\]
and $k_B T=(1.38\times 10^{-23}\,\text{J/K})(0.05\,\text{K})
\approx 6.9\times 10^{-25}\,\text{J}$. Hence
$\hbar\omega_q/(k_B T)\approx 4.8$ and
$p_1/p_0\approx e^{-4.8}\approx 8\times 10^{-3}$, i.e.\ the
\emph{thermal} excited-state population is at the per-cent level
in equilibrium with a 50 mK bath. In practice, residual
populations in real devices are limited not by this thermal
ratio but by stray IR radiation, quasiparticle poisoning, and
non-equilibrium photon noise; cQED experiments accordingly
combine cryogenic operation with active reset protocols when
better-than-thermal initialisation is required.
\end{solution}
\end{exercise}
