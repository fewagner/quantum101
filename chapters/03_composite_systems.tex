\chapter{Composite systems, density matrices and entanglement}
\label{ch:composite}

\section*{Chapter overview}

Up to this point we have described isolated systems with a single
ket. To talk about a qubit coupled to a resonator, two qubits in a
gate, or a qubit interacting with its environment, we need three
pieces of new machinery: (i) the tensor product, which combines two
Hilbert spaces into one; (ii) the density matrix, which extends the
notion of state to situations with classical uncertainty or
correlations with inaccessible degrees of freedom; and (iii) the
partial trace, which extracts the effective state of a subsystem from
that of a larger composite. With these tools we will see that
entanglement --- which has no classical analogue --- is the generic
behaviour of interacting quantum systems and the resource that powers
both quantum information processing and the standard tomography
techniques used in cQED.

\section{Composite systems and tensor products}
\label{sec:tensor}

\begin{phenomenon}
Two independent quantum systems prepared in states $\ket{\psi_A}$
and $\ket{\psi_B}$ are jointly described by a single state in a
larger Hilbert space, the tensor product $\hilbert_A\otimes\hilbert_B$.
This much is intuitive: two non-interacting qubits, for instance,
can clearly be described by listing the state of each. The
genuinely new feature is that the tensor product is \emph{larger}
than the set of such product descriptions: it contains many states
that cannot be written as $\ket{\psi_A}\otimes\ket{\psi_B}$ for
\emph{any} choice of $\ket{\psi_A}$ and $\ket{\psi_B}$. Such states
are called \emph{entangled}; they encode correlations between the
two subsystems that have no classical analogue. The example below
shows concretely that even one of the simplest-looking two-qubit
states, $\tfrac{1}{\sqrt 2}(\ket{00}+\ket{11})$, cannot be
factorised --- any attempt to write it as a product of
single-qubit states leads to a direct contradiction.
\end{phenomenon}

\begin{postulate}[Composite systems]
\label{post:composite}
The Hilbert space of a composite system $AB$ is the tensor product
$\hilbert_{AB}=\hilbert_A\otimes\hilbert_B$
(\cref{def:tensor-product}). If $\{\ket{i}_A\}$ and $\{\ket{j}_B\}$
are orthonormal bases of $\hilbert_A,\hilbert_B$, then
$\{\ket{i}_A\otimes\ket{j}_B\}$ is an orthonormal basis of
$\hilbert_{AB}$. Operators acting only on $A$ or $B$ are extended by
the identity: $\op{O}_A\to\op{O}_A\otimes\id_B$,
$\op{O}_B\to\id_A\otimes\op{O}_B$.
\end{postulate}

\begin{example}[Two qubits and the Bell states]
\label{ex:bell-states}
For two qubits, $\hilbert_{AB}=\C^2\otimes\C^2=\C^4$ with basis
$\{\ket{00},\ket{01},\ket{10},\ket{11}\}$. The four \emph{Bell
states}
\begin{equation}
\ket{\Phi^\pm} = \tfrac{1}{\sqrt2}(\ket{00}\pm\ket{11}),
\qquad
\ket{\Psi^\pm} = \tfrac{1}{\sqrt2}(\ket{01}\pm\ket{10}),
\end{equation}
form a maximally entangled basis of $\C^4$: none of them is a
product. Let us verify this for $\ket{\Phi^+}$ by direct
calculation. Suppose, for contradiction, that
\begin{equation}
\ket{\Phi^+} \;=\;
(\alpha\ket{0}+\beta\ket{1})\otimes(\gamma\ket{0}+\delta\ket{1})
\;=\; \alpha\gamma\,\ket{00} + \alpha\delta\,\ket{01}
   + \beta\gamma\,\ket{10} + \beta\delta\,\ket{11},
\end{equation}
for some $\alpha,\beta,\gamma,\delta\in\C$. Matching coefficients
against $\ket{\Phi^+}=\tfrac{1}{\sqrt2}(\ket{00}+\ket{11})$ gives
\begin{equation}
\alpha\gamma=\tfrac{1}{\sqrt2},\quad
\beta\delta=\tfrac{1}{\sqrt2},\quad
\alpha\delta=0,\quad
\beta\gamma=0.
\end{equation}
The first two equations require $\alpha,\gamma,\beta,\delta$ all
non-zero; the last two then force $\alpha\delta=0$, a
contradiction. There are no single-qubit states whose tensor
product is $\ket{\Phi^+}$, so $\ket{\Phi^+}$ is genuinely
entangled. The same argument disposes of the other three Bell
states.
\end{example}

\paragraph{Transition.}
The four Bell states already exhibit the central oddity of composite
systems: a perfectly known joint state can correspond to two
\emph{individually} undefined subsystem states. To make sense of this,
we now generalise the notion of state itself.

\section{Density matrices}
\label{sec:density}

\begin{phenomenon}
We rarely know the exact pure state of a real system: the
preparation procedure may produce $\ket{\psi_n}$ with classical
probability $p_n$, the system may be entangled with a bath, or the
experimenter may simply have a coarse-grained description of the
state. In all such cases the relevant object is a probability-weighted
mixture of pure states, the density matrix.
\end{phenomenon}

\paragraph{Beyond the qubit case.}
We already met the density matrix as the right description of a
single qubit when its preparation is uncertain
(\cref{def:density-tls} in \cref{sec:bloch}). The construction
makes sense in any Hilbert space, not just $\C^2$, and we restate
it here in full generality. The defining conditions
\emph{Hermitian, positive, unit-trace} are unchanged from
\cref{def:density-tls}; what is new is that $\hilbert$ is now
arbitrary --- in particular, it can be a tensor product
$\hilbert_A\otimes\hilbert_B$ of two subsystems, the case which
will let us describe a qubit's effective state when it is
entangled with another system. The single-qubit Bloch
parametrisation (\cref{stmt:bloch}) is the
$\hilbert=\C^2$ specialisation of what follows.

\begin{definition}[Density matrix, general]
\label{def:density}
A \emph{density matrix} on a Hilbert space $\hilbert$ is a linear
operator $\op{\rho}:\hilbert\to\hilbert$ such that
$\op{\rho}^\dagger=\op{\rho}$, $\op{\rho}\geq 0$, and
$\Tr\op{\rho}=1$. The set of all such operators is convex; its
extreme points are the \emph{pure states}
$\op{\rho}=\ket{\psi}\bra{\psi}$ and all other density matrices are
\emph{mixed}.
\end{definition}

For a statistical mixture of pure states $\{(\ket{\psi_n},p_n)\}$ the
density matrix is
\begin{equation}
\op{\rho} \;=\; \sum_n p_n\,\ket{\psi_n}\bra{\psi_n},
\qquad p_n\geq0,\;\sum_n p_n=1.
\end{equation}
The purity introduced for the qubit case in \cref{def:density-tls}
extends without change: $\Tr\op{\rho}^2=1$ exactly when $\op{\rho}$
is pure, and $\Tr\op{\rho}^2<1$ otherwise.

\begin{statement}[Born rule and dynamics, density-matrix form]
\label{stmt:rho-rules}
The expectation value of an observable $\op{A}$ in state $\op{\rho}$
is
\begin{equation}
\expval{\op{A}} \;=\; \Tr(\op{\rho}\,\op{A}),
\end{equation}
and unitary evolution under a Hamiltonian $\op{H}$ obeys the
Liouville--von~Neumann equation
\begin{equation}
\label{eq:liouville}
i\hbar\,\dv{t}\op{\rho} \;=\; [\op{H},\op{\rho}].
\end{equation}
\end{statement}

\begin{proof}
For a pure state $\ket{\psi}$, $\Tr(\op{\rho}\op{A})=\bra{\psi}\op{A}\ket{\psi}$
by inserting the resolution of identity; the result extends by
linearity to mixtures. For dynamics,
$\op{\rho}(t)=\op{U}\op{\rho}(0)\op{U}^\dagger$ with
$\op{U}=e^{-i\op{H}t/\hbar}$; differentiating gives
\cref{eq:liouville}.
\end{proof}

\paragraph{Connection to the Schr\"odinger equation.}
The dynamical part of \cref{stmt:rho-rules} is the Schr\"odinger
equation rewritten in density-matrix language. A pure-state
evolution under a Hamiltonian $\op{H}$ is
$\ket{\psi(t)}=\op{U}(t)\ket{\psi(0)}$ with
$\op{U}(t)=e^{-i\op{H}t/\hbar}$ (\cref{post:schrodinger}). The
corresponding density matrix therefore evolves as
\begin{equation}
\op{\rho}(t) \;=\; \ket{\psi(t)}\bra{\psi(t)}
              \;=\; \op{U}(t)\,\op{\rho}(0)\,\op{U}^\dagger(t),
\end{equation}
and differentiating both sides while using
$i\hbar\,\dot{\op{U}}=\op{H}\op{U}$ gives the Liouville--von~Neumann
equation \cref{eq:liouville}. By linearity the same equation holds
for any \emph{mixture} $\op{\rho}=\sum_n p_n\,\ket{\psi_n}\bra{\psi_n}$
of pure states evolving under the same $\op{H}$, since each
$\ket{\psi_n}\bra{\psi_n}$ obeys it term by term.

So the Liouville--von~Neumann equation is not a new dynamical
postulate: it is the Schr\"odinger equation expressed in the
language that also handles classical uncertainty in the
preparation. The two are strictly equivalent for an isolated
system. What is genuinely new comes in Part~II: when the system is
\emph{open} (coupled to an environment) we cannot describe its
dynamics by a single Hamiltonian, and the density matrix evolves
under a generally non-unitary equation --- the Lindblad master
equation --- that has no analogue at the level of a state vector,
because the state can stop being pure even if it started so.

\paragraph{Transition.}
A density matrix on $\hilbert_{AB}$ describes the joint state. The
question of what state subsystem $A$ effectively sees, when $B$ is
inaccessible, is answered by the partial trace.

\section{The partial trace and reduced states}
\label{sec:partial-trace}

\begin{definition}[Partial trace]
\label{def:partial-trace}
Let $\op{\rho}_{AB}$ be a density matrix on
$\hilbert_A\otimes\hilbert_B$ and $\{\ket{j}_B\}$ a basis of
$\hilbert_B$. The \emph{reduced density matrix} of $A$ is
\begin{equation}
\label{eq:partial-trace}
\op{\rho}_A \;=\; \Tr_B\,\op{\rho}_{AB}
\;=\; \sum_j \bra{j}_B\,\op{\rho}_{AB}\,\ket{j}_B.
\end{equation}
$\op{\rho}_A$ is the unique operator on $\hilbert_A$ such that
$\Tr_A(\op{\rho}_A\,\op{O}_A)=\Tr(\op{\rho}_{AB}\,(\op{O}_A\otimes\id_B))$
for every observable $\op{O}_A$ on $A$.
\end{definition}

\begin{statement}[Reduction of an entangled pure state]
\label{stmt:reduced-pure}
Let $\ket{\Psi}_{AB}$ be a pure state. Its reduced state
$\op{\rho}_A=\Tr_B\,\ket{\Psi}\bra{\Psi}$ is mixed if and only if
$\ket{\Psi}$ is entangled. In particular,
\begin{equation}
\Tr\op{\rho}_A^2 = 1 \;\;\Longleftrightarrow\;\;
\ket{\Psi}_{AB} = \ket{\psi}_A\otimes\ket{\phi}_B
\text{ for some }\ket{\psi},\ket{\phi}.
\end{equation}
\end{statement}

\begin{proof}[Sketch via Schmidt decomposition]
Any bipartite pure state admits the Schmidt decomposition
$\ket{\Psi}=\sum_k \lambda_k\,\ket{u_k}_A\otimes\ket{v_k}_B$ with
$\lambda_k\geq0$ and $\sum_k\lambda_k^2=1$. Tracing over $B$ yields
$\op{\rho}_A=\sum_k\lambda_k^2\,\ket{u_k}\bra{u_k}$, which is pure
($\lambda_k=\delta_{k1}$) precisely when only one Schmidt coefficient
is non-zero, i.e.\ when $\ket{\Psi}$ factorises.
\end{proof}

\begin{example}[Bell state has maximally mixed marginals]
For $\ket{\Phi^+}=(\ket{00}+\ket{11})/\sqrt2$ a direct calculation
gives
$\op{\rho}_A = \tfrac12(\ket{0}\bra{0}+\ket{1}\bra{1}) = \tfrac12\id$.
Each qubit individually is in the maximally mixed state $\id/2$ ---
its Bloch vector is the origin of the Bloch sphere --- even though
the joint state is perfectly known.
\end{example}

\begin{example}[Decoherence as system--environment entanglement]
\label{ex:decoherence-by-partial-trace}
The Bell-state example above already showed that tracing out a
subsystem can turn a pure joint state into a mixed marginal. That
mechanism is the seed of \emph{decoherence} --- the loss of
quantum coherence due to interaction with an environment --- which
we treat systematically in \cref{ch:open-systems}. Here we exhibit
it in its simplest possible form: a single environment qubit and a
single round of interaction.

\paragraph{Setup.}
Take a system qubit $S$ in the equal superposition
\begin{equation}
\ket{\psi(0)}_S \;=\; \ket{+}_S
\;\equiv\; \tfrac{1}{\sqrt 2}\bigl(\ket{0}+\ket{1}\bigr),
\end{equation}
and an environment qubit $E$ in $\ket{0}_E$. The joint state is a
product, hence separable:
\begin{equation}
\ket{\Psi(0)}_{SE}
\;=\; \ket{+}_S\!\otimes\!\ket{0}_E
\;=\; \tfrac{1}{\sqrt 2}\bigl(\ket{00}+\ket{10}\bigr).
\end{equation}
The reduced state of $S$ is
$\op{\rho}_S(0)=\ket{+}\bra{+}=\tfrac12(\id+\hat\sigma_x)$, with
Bloch vector $\bm{r}(0)=(1,0,0)$ on the equator of the Bloch
sphere; in particular $\op{\rho}_S(0)$ is pure and has off-diagonal
coherence $\bra{0}\op{\rho}_S\ket{1}=\tfrac12$.

\paragraph{Interaction.}
Suppose $S$ and $E$ now interact through a unitary $\op{U}_{SE}$
that copies the system's computational-basis state onto the
environment,
\begin{equation}
\op{U}_{SE}\,\ket{0}_S\ket{0}_E = \ket{0}_S\ket{0}_E,
\qquad
\op{U}_{SE}\,\ket{1}_S\ket{0}_E = \ket{1}_S\ket{1}_E.
\end{equation}
(This is the action of a controlled-NOT gate; we encounter the
gate again in \cref{ex:bell-states} as a way of \emph{producing}
$\ket{\Phi^+}$, and its hardware realisation is in
\cref{sec:two-qubit-gates}.) By linearity, applied to
$\ket{\Psi(0)}_{SE}$ it gives
\begin{equation}
\ket{\Psi}_{SE}
\;=\; \tfrac{1}{\sqrt 2}\bigl(\ket{00}+\ket{11}\bigr)
\;=\; \ket{\Phi^+}.
\end{equation}
The joint state is still pure, the dynamics is unitary, and \emph{no
information has been lost from the universe}.

\paragraph{What the system sees.}
Tracing out the environment, the reduced state of $S$ is
\begin{equation}
\op{\rho}_S \;=\; \Tr_E\,\ket{\Phi^+}\bra{\Phi^+}
              \;=\; \tfrac12 \id.
\end{equation}
The Bloch vector has contracted from $(1,0,0)$ to the origin. The
diagonal populations of $\op{\rho}_S$ are unchanged ($\tfrac12$ at
both ends), but the off-diagonal coherence
$\bra{0}\op{\rho}_S\ket{1}$ has gone from $\tfrac12$ to $0$. This
process --- destruction of coherence without exchange of populations
--- is \emph{pure dephasing}.

\paragraph{Decoherence is hidden information.}
The off-diagonal coherence was not destroyed: it was \emph{moved}
into correlations between $S$ and $E$ that the partial trace
discards. With access to $E$ we could in principle reverse the
interaction (it is unitary) and recover the original pure state of
$S$. We do not have that access; from $S$'s point of view the
coherence is gone.

\paragraph{Forward to Part II.}
A real cQED environment is not a single qubit but a continuum of
modes with effectively random phases (a transmission line, a phonon
bath, two-level fluctuators), continuously interacting with the
system. The same mechanism then produces a smooth exponential decay
of the off-diagonal density-matrix elements, with characteristic
timescale $T_2$. The systematic theory --- the Born--Markov
approximation, the Lindblad master equation, the relaxation
($T_1$), dephasing ($T_2$) and pure-dephasing ($T_2^*$) times --- is
\cref{ch:open-systems}. The example above is its kernel in three
lines.
\end{example}

\begin{remark}
The same construction underlies the description of an open system
coupled to a bath: the system's effective state is the partial trace
of the system--bath density matrix over the bath. The Lindblad
master equation of \cref{ch:open-systems} is the time-local equation
this reduced state obeys under the Born--Markov approximation.
\end{remark}

\paragraph{Transition.}
Entanglement, having appeared as the obstruction to writing
$\ket{\Psi}$ as a product, deserves a section of its own --- both as
a phenomenon and as a quantitative resource.

\section{Entanglement}
\label{sec:entanglement}

\begin{definition}[Separable and entangled states]
\label{def:entanglement}
A pure bipartite state $\ket{\Psi}_{AB}$ is \emph{separable} if it
factorises as $\ket{\psi}_A\otimes\ket{\phi}_B$, otherwise it is
\emph{entangled}. A mixed state $\op{\rho}_{AB}$ is separable if it
admits a decomposition
$\op{\rho}_{AB}=\sum_n p_n\,\op{\rho}_A^{(n)}\otimes\op{\rho}_B^{(n)}$
with $p_n\geq0$, $\sum_n p_n=1$, and entangled otherwise.
\end{definition}

\begin{statement}[Schmidt rank as an entanglement witness]
\label{stmt:schmidt}
The Schmidt rank of a bipartite pure state $\ket{\Psi}_{AB}$ ---
i.e.\ the number of non-zero Schmidt coefficients $\lambda_k$ in
$\ket{\Psi}=\sum_k\lambda_k\ket{u_k}_A\ket{v_k}_B$ --- equals one if
and only if $\ket{\Psi}$ is separable. Larger Schmidt rank means
more entanglement; the entropy $S(\op{\rho}_A)=-\sum_k\lambda_k^2\log\lambda_k^2$
of the reduced state is the canonical quantitative measure.
\end{statement}

\begin{example}[Two-qubit entanglement: one ``ebit'' in $\ket{\Phi^+}$]
We compute the entanglement quantitatively for the Bell state of
\cref{ex:bell-states}. By inspection, $\ket{\Phi^+}$ is already in
Schmidt form,
\begin{equation}
\label{eq:phiplus-schmidt}
\ket{\Phi^+} \;=\; \tfrac{1}{\sqrt2}\,\ket{0}_A\!\otimes\!\ket{0}_B
                 + \tfrac{1}{\sqrt2}\,\ket{1}_A\!\otimes\!\ket{1}_B,
\end{equation}
with two non-zero Schmidt coefficients
$\lambda_1=\lambda_2=\tfrac{1}{\sqrt2}$. Because the two
$\lambda_k$ are equal, the bipartite state is \emph{maximally
entangled} on $\C^2\otimes\C^2$: any other choice of unit-norm
$\lambda_k$ with $\lambda_1^2+\lambda_2^2=1$ has at least one
coefficient strictly larger than $1/\sqrt 2$ and is more
``product-like'' in the sense that subsystem $A$ retains more
information about its own state.

\paragraph{The reduced state.}
Applying the partial-trace formula \cref{eq:partial-trace} to
\cref{eq:phiplus-schmidt} (or specialising the proof of
\cref{stmt:reduced-pure} to this case) gives
\begin{equation}
\op{\rho}_A \;=\; \Tr_B\,\ket{\Phi^+}\bra{\Phi^+}
              \;=\; \lambda_1^2\,\ket{0}\bra{0}
                  + \lambda_2^2\,\ket{1}\bra{1}
              \;=\; \tfrac12\,\id,
\end{equation}
i.e.\ qubit $A$ on its own is in the maximally mixed state ---
Bloch vector at the origin. Even though the joint $\ket{\Phi^+}$
is perfectly known, an observer with access only to $A$ has zero
predictive information about $A$'s observables: this is the
content of \cref{stmt:reduced-pure} made concrete.

\paragraph{The entanglement entropy.}
Plug $\lambda_1=\lambda_2=1/\sqrt 2$ into the entropy formula of
\cref{stmt:schmidt}:
\begin{equation}
S(\op{\rho}_A)
\;=\; -\sum_k \lambda_k^2\,\log_2 \lambda_k^2
\;=\; -2\cdot\tfrac12\,\log_2\tfrac12
\;=\; \log_2 2 \;=\; 1\;\text{bit}.
\end{equation}
This is the maximum value $S(\op{\rho}_A)$ can take on
$\C^2\otimes\C^2$ (it is bounded by $\log_2\dim\hilbert_A=1$ for a
qubit), confirming that $\ket{\Phi^+}$ is maximally entangled.

\paragraph{The unit ``ebit''.}
The result $S(\op{\rho}_A)=1$ bit motivates a unit of its own. An
\emph{ebit} (entanglement bit) is the conventional unit of
bipartite pure-state entanglement, defined as the amount of
entanglement contained in one Bell pair: a state with
$S(\op{\rho}_A)=n$ bits carries $n$ ebits. The unit is more than
notational. Two protocols give it operational meaning --- one
that \emph{consumes} ebits and one that \emph{produces} them ---
and they let us introduce two pieces of quantum-information
terminology that will recur in later parts.

\paragraph{Where ebits are spent: quantum teleportation.}
\emph{Quantum teleportation} is a protocol in which Alice
transmits an unknown one-qubit state to Bob using only (i) one
pre-shared Bell pair and (ii) two classical bits sent over a
classical channel; no qubit physically travels from Alice to
Bob. Each successful teleportation \emph{consumes} the shared
Bell pair, i.e.\ exactly one ebit. The protocol thus turns ``one
ebit + two classical bits'' into ``one transmitted qubit'', and
is the canonical example of a task whose cost is an integer
number of ebits. We treat it carefully in \cref{ch:qec}; the
point here is only that ebits are a real, expendable resource.

\paragraph{Where ebits are made: two-qubit entangling gates and CNOT.}
A \emph{two-qubit gate} is a unitary on
$\hilbert_A\otimes\hilbert_B$ that \emph{does not} factorise as a
product $\op{U}_A\otimes\op{U}_B$ of single-qubit unitaries.
Equivalently (\cref{app:entanglement-cqed}), it is a gate capable
of taking a separable input to an entangled output. The canonical
example, already used informally in
\cref{ex:decoherence-by-partial-trace} above, is the
\emph{controlled-NOT} or \emph{CNOT} gate: it flips the target
qubit $B$ if and only if the control qubit $A$ is in $\ket{1}$,
\begin{equation}
\text{CNOT}\,\ket{a}_A\ket{b}_B
\;=\; \ket{a}_A\,\ket{a\oplus b}_B,
\end{equation}
with $\oplus$ addition modulo $2$. Acting on the separable input
$\ket{+}_A\otimes\ket{0}_B$ produces the maximally entangled
$\ket{\Phi^+}$,
\begin{equation}
\text{CNOT}\,\bigl(\tfrac{1}{\sqrt 2}(\ket{0}+\ket{1})_A\!\otimes\!\ket{0}_B\bigr)
\;=\; \tfrac{1}{\sqrt 2}(\ket{00}+\ket{11})
\;=\; \ket{\Phi^+},
\end{equation}
i.e.\ generates one ebit from a separable input. Hardware
realisations of CNOT-equivalent gates in superconducting circuits
(cross-resonance, iSWAP-based, and so on) are the subject of
\cref{sec:two-qubit-gates}.
\end{example}

\begin{application}[Why entanglement matters in cQED]
\label{app:entanglement-cqed}
Two-qubit gates (\cref{sec:two-qubit-gates}) are characterised by
their ability to entangle: a gate that cannot create entanglement is
locally equivalent to a product of single-qubit gates and provides
no quantum-computational power. Conversely, entanglement with the
environment is the cause of decoherence
(\cref{ch:open-systems}): tracing out the environment in an
entangled system--bath state produces a mixed reduced state for the
qubit.
\end{application}

\section*{Chapter takeaways and outlook}

\begin{chaptersummary}

\paragraph{Composite systems.}
The Hilbert space of a composite system $AB$ is the tensor product
$\hilbert_{AB}=\hilbert_A\otimes\hilbert_B$. Operators on a single
subsystem are extended by the identity:
$\op{O}_A\to\op{O}_A\otimes\id_B$,
$\op{O}_B\to\id_A\otimes\op{O}_B$. The four Bell states
\begin{equation}
\ket{\Phi^\pm}=\tfrac{1}{\sqrt 2}(\ket{00}\pm\ket{11}),\quad
\ket{\Psi^\pm}=\tfrac{1}{\sqrt 2}(\ket{01}\pm\ket{10})
\end{equation}
form a maximally entangled basis of $\C^2\otimes\C^2$.

\paragraph{Density matrix (general).}
A density matrix on $\hilbert$ is Hermitian, positive semi-definite,
and trace-one. Pure states are
$\op{\rho}=\ket{\psi}\bra{\psi}$ (purity $\Tr\op{\rho}^2=1$);
all other density matrices are mixed. Born rule and unitary
evolution take their density-matrix forms,
\begin{equation}
\expval{\op{A}} \;=\; \Tr(\op{\rho}\,\op{A}),
\qquad
i\hbar\,\dv{t}\op{\rho} \;=\; [\op{H},\op{\rho}],
\end{equation}
the latter (Liouville--von~Neumann) being the Schr\"odinger
equation rewritten for mixtures. For an isolated system the two
formulations are strictly equivalent.

\paragraph{Partial trace.}
For $\op{\rho}_{AB}$ on $\hilbert_A\otimes\hilbert_B$, the reduced
density matrix of $A$ is
\begin{equation}
\op{\rho}_A \;=\; \Tr_B\,\op{\rho}_{AB}
\;=\; \sum_j\bra{j}_B\,\op{\rho}_{AB}\,\ket{j}_B.
\end{equation}
For a pure $\ket{\Psi}_{AB}$, the reduced state is mixed if and
only if $\ket{\Psi}$ is entangled, sharpened by the Schmidt
decomposition $\ket{\Psi}=\sum_k\lambda_k\ket{u_k}_A\ket{v_k}_B$.

\paragraph{Entanglement and the ebit.}
A pure bipartite state is separable iff its Schmidt rank is one.
The canonical quantitative measure is the entanglement entropy of
the reduced state,
\begin{equation}
S(\op{\rho}_A) \;=\; -\sum_k\lambda_k^2\,\log_2\lambda_k^2,
\end{equation}
in units of \emph{ebits}. One ebit is the entanglement contained
in a single Bell pair. Two-qubit entangling gates such as CNOT
generate ebits from separable inputs; quantum teleportation
consumes them.

\paragraph{Decoherence in three lines.}
A system qubit in $\ket{+}_S$ plus an environment qubit in
$\ket{0}_E$ becomes the joint Bell state $\ket{\Phi^+}$ after a
state-copy interaction; tracing out $E$ leaves the system in
$\op{\rho}_S=\id/2$, with its Bloch vector contracted from
$(1,0,0)$ to the origin. The off-diagonal coherence has not been
destroyed but transferred into $S$--$E$ correlations that we
discard. This is the kernel of pure dephasing; the
continuous-bath generalisation gives the smooth $T_2$ decay
treated systematically in Part~II.

\end{chaptersummary}

\paragraph{Outlook.}
The final chapter of this part deals with one further question
that the postulates do not, by themselves, answer: how does a
system respond to a small or near-resonant time-dependent drive?
The answer assembles two of the workhorses of quantum optics ---
time-dependent perturbation theory (with Fermi's golden rule) and
the rotating-wave approximation --- both of which we will use
repeatedly in Parts~III and IV.

\begin{exercise}[Bell states are orthonormal]
Verify that the four Bell states
$\ket{\Phi^\pm},\ket{\Psi^\pm}$ form an orthonormal basis of
$\C^2\otimes\C^2$.
\begin{solution}
Each Bell state has norm $1$ by inspection of the
$1/\sqrt 2$ coefficients. Pairwise:
$\braket{\Phi^+}{\Phi^-}=\tfrac12(\braket{00}{00}-\braket{11}{11})=0$;
$\braket{\Phi^+}{\Psi^+}=\tfrac12(\braket{00}{01}+\braket{11}{10})=0$;
and so on for the remaining pairs --- in each case the four cross
terms either pair $\ket{0i}$ with $\ket{0j}$ ($i\neq j$) or
$\ket{1i}$ with $\ket{1j}$ ($i\neq j$), all of which vanish.
Hence orthogonal, and together with normalisation an orthonormal
basis of $\C^4$.
\end{solution}
\end{exercise}

\begin{exercise}[Partial trace of a product state]
Compute $\Tr_B\,\bigl(\ket{\psi}_A\bra{\psi}\otimes\ket{\phi}_B\bra{\phi}\bigr)$
explicitly. Confirm the result is $\ket{\psi}_A\bra{\psi}$.
\begin{solution}
Write $\ket{\phi}_B=\sum_j c_j\ket{j}_B$ in an orthonormal basis,
so $\ket{\phi}\bra{\phi}=\sum_{jk}c_j\bar c_k\ket{j}\bra{k}$. Then
$\Tr_B\bigl(\ket{\phi}\bra{\phi}\bigr)=\sum_l\bra{l}\bigl(\sum_{jk}c_j\bar c_k\ket{j}\bra{k}\bigr)\ket{l}
=\sum_l|c_l|^2=\braket{\phi}{\phi}=1$.
Hence the partial trace is
$\ket{\psi}_A\bra{\psi}\cdot 1=\ket{\psi}_A\bra{\psi}$, as a product
state has product reduced states.
\end{solution}
\end{exercise}

\begin{exercise}[Purity of the maximally mixed state]
Compute the purity $\Tr\op{\rho}^2$ of $\op{\rho}=\id_d/d$ on a
$d$-dimensional Hilbert space.
\begin{solution}
$\op{\rho}^2=\id_d/d^2$, so
$\Tr\op{\rho}^2=\Tr\id_d/d^2=d/d^2=1/d$. For a qubit
($d=2$) this is $1/2$, the minimum possible purity for a single
qubit and the operational meaning of ``maximally mixed''.
\end{solution}
\end{exercise}

\begin{exercise}[Schmidt decomposition of $\ket{\Psi^+}$]
Write $\ket{\Psi^+}=(\ket{01}+\ket{10})/\sqrt 2$ in Schmidt form.
What are the Schmidt coefficients? Compute the entanglement
entropy $S(\op{\rho}_A)$.
\begin{solution}
$\ket{\Psi^+}=\tfrac{1}{\sqrt 2}\ket{0}_A\!\otimes\!\ket{1}_B
+\tfrac{1}{\sqrt 2}\ket{1}_A\!\otimes\!\ket{0}_B$, already in
Schmidt form with $\lambda_1=\lambda_2=1/\sqrt 2$. Hence
$\op{\rho}_A=\tfrac12\ket{0}\bra{0}+\tfrac12\ket{1}\bra{1}=\id/2$,
and $S(\op{\rho}_A)=-2\cdot\tfrac12\log_2\tfrac12=1$ bit ($1$
ebit) --- maximally entangled, like all four Bell states.
\end{solution}
\end{exercise}

\begin{exercise}[Partially-entangled two-qubit state]
For
$\ket{\Psi}_{AB}=\alpha\ket{00}+\beta\ket{11}$ with
$|\alpha|^2+|\beta|^2=1$, compute $\op{\rho}_A$ and
$S(\op{\rho}_A)$. For what choice of $|\alpha|$ is the entanglement
entropy maximal? minimal?
\begin{solution}
$\ket\Psi$ is in Schmidt form with $\lambda_1=|\alpha|$,
$\lambda_2=|\beta|$, so
$\op{\rho}_A=|\alpha|^2\ket{0}\bra{0}+|\beta|^2\ket{1}\bra{1}$.
Hence $S(\op{\rho}_A)=-|\alpha|^2\log_2|\alpha|^2-|\beta|^2\log_2|\beta|^2
=h(|\alpha|^2)$, the binary Shannon entropy. It is maximal
($S=1$ ebit) for $|\alpha|=|\beta|=1/\sqrt 2$, minimal ($S=0$,
separable) for $|\alpha|\in\{0,1\}$.
\end{solution}
\end{exercise}

\begin{exercise}[Bell-state correlations]
For the Bell state $\ket{\Phi^+}$, compute the joint expectation
values $\langle\hat\sigma_z\otimes\hat\sigma_z\rangle$,
$\langle\hat\sigma_x\otimes\hat\sigma_x\rangle$, and
$\langle\hat\sigma_y\otimes\hat\sigma_y\rangle$. Comment on the
sign of each.
\begin{solution}
In the computational basis,
$(\hat\sigma_z\otimes\hat\sigma_z)\ket{ij}=(-1)^{i+j}\ket{ij}$, so
$\langle\hat\sigma_z\otimes\hat\sigma_z\rangle_{\Phi^+}
=\tfrac12(+1)+\tfrac12(+1)=+1$ (perfect correlation: equal $z$-outcomes).
$(\hat\sigma_x\otimes\hat\sigma_x)\ket{ij}=\ket{(1-i)(1-j)}$, so
acting on $\ket{\Phi^+}$ swaps $\ket{00}\leftrightarrow\ket{11}$,
giving $+1$ (perfect correlation in $x$).
$(\hat\sigma_y\otimes\hat\sigma_y)\ket{ij}=(-1)^{i+j}(i)^2\ket{(1-i)(1-j)}$;
acting on $\ket{\Phi^+}$ gives $-1$ (perfect anti-correlation in
$y$). The pattern $\langle XX\rangle=+1$, $\langle YY\rangle=-1$,
$\langle ZZ\rangle=+1$ is the diagonal signature of $\ket{\Phi^+}$.
\end{solution}
\end{exercise}

\begin{exercise}[Liouville--von Neumann from the Schr\"odinger equation]
Derive \cref{eq:liouville} by differentiating
$\op{\rho}(t)=\op{U}(t)\,\op{\rho}(0)\,\op{U}^\dagger(t)$ and using
$i\hbar\,\dot{\op{U}}=\op{H}\op{U}$.
\begin{solution}
$\dv{t}\op{\rho}=\dot{\op{U}}\op{\rho}_0\op{U}^\dagger
+\op{U}\op{\rho}_0\dot{\op{U}}^\dagger$. Using
$\dot{\op{U}}=-\tfrac{i}{\hbar}\op{H}\op{U}$ and the Hermitian
conjugate $\dot{\op{U}}^\dagger=+\tfrac{i}{\hbar}\op{U}^\dagger\op{H}$,
\[
\dv{t}\op{\rho}=-\tfrac{i}{\hbar}\op{H}\op{U}\op{\rho}_0\op{U}^\dagger
+\tfrac{i}{\hbar}\op{U}\op{\rho}_0\op{U}^\dagger\op{H}
=-\tfrac{i}{\hbar}\bigl(\op{H}\op{\rho}-\op{\rho}\op{H}\bigr)
=-\tfrac{i}{\hbar}[\op{H},\op{\rho}],
\]
which rearranges to \cref{eq:liouville}.
\end{solution}
\end{exercise}

\begin{exercise}[Measurement backaction on a Bell pair]
Alice holds qubit $A$ of $\ket{\Phi^+}_{AB}$ and measures
$\hat\sigma_x$, obtaining the outcome $+1$. What is the
post-measurement state of qubit $B$?
\begin{solution}
Rewrite $\ket{\Phi^+}=(\ket{++}+\ket{--})/\sqrt 2$, where
$\ket{\pm}=(\ket{0}\pm\ket{1})/\sqrt 2$ are the
$\hat\sigma_x$-eigenstates. The projection onto Alice's $+$
eigenspace is $\op{P}^A_+\otimes\id_B
=\ket{+}_A\bra{+}\otimes\id_B$. Acting on $\ket{\Phi^+}$ gives
$\tfrac{1}{\sqrt 2}\ket{+}_A\!\otimes\!\ket{+}_B$, normalised:
$\ket{+}_A\!\otimes\!\ket{+}_B$. So conditional on Alice's $+1$
result, qubit $B$ is in $\ket{+}_B$. Measurement on Alice's qubit
remotely projects qubit $B$ onto a definite state --- the
classical-information channel that quantum teleportation
formalises.
\end{solution}
\end{exercise}

\begin{exercise}[Coherence vs.\ population in pure dephasing]
A qubit's density matrix in the energy basis is
$\op{\rho}=\bigl(\begin{smallmatrix}p&c\\\bar c&1-p\end{smallmatrix}\bigr)$
with $p\in[0,1]$ and $|c|\leq\sqrt{p(1-p)}$. Show that pure
dephasing ($\hat\sigma_z$-noise on the system) can reduce $|c|$
to $0$ \emph{without} changing $p$. What does this look like on
the Bloch sphere?
\begin{solution}
Pure dephasing implements a phase-flip channel
$\op{\rho}\to(1-q)\op{\rho}+q\,\hat\sigma_z\op{\rho}\hat\sigma_z$.
$\hat\sigma_z\op{\rho}\hat\sigma_z=\bigl(\begin{smallmatrix}p&-c\\-\bar c&1-p\end{smallmatrix}\bigr)$,
so the diagonal entries are unchanged and the off-diagonal becomes
$c\to(1-2q)c$. For $q=1/2$ the off-diagonal vanishes; the
populations $p,1-p$ are untouched. On the Bloch sphere this
contracts the $(r_x,r_y)$ components to zero while leaving $r_z$
fixed --- the Bloch vector slides toward the $z$-axis.
\end{solution}
\end{exercise}

\begin{exercise}[Reduced state of an entangled three-qubit GHZ]
Compute $\op{\rho}_A=\Tr_{BC}\,\ket{\text{GHZ}}\bra{\text{GHZ}}$
for $\ket{\text{GHZ}}=(\ket{000}+\ket{111})/\sqrt 2$. What is the
purity of $\op{\rho}_A$?
\begin{solution}
Tracing out qubits $B$ and $C$ in the computational basis,
$\op{\rho}_A=\sum_{jk}\bra{jk}_{BC}(\ket{\text{GHZ}}\bra{\text{GHZ}})\ket{jk}_{BC}
=\tfrac12\ket{0}\bra{0}+\tfrac12\ket{1}\bra{1}=\id/2$. Purity
$\Tr\op{\rho}_A^2=1/2$, the same as for a Bell-pair marginal:
qubit $A$ is in the maximally mixed state, despite being only one
of three constituents.
\end{solution}
\end{exercise}
